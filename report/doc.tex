\section{اهداف و دامنه پژوهش}

هدف اصلی این پژوهش آن است که بررسی شود آیا عامل‌های \textbf{یادگیری تقویتی}\footnote{Reinforcement Learning} که با اطلاعات احساسی استخراج‌شده از \textbf{مدل‌های زبانی بزرگ}\footnote{Large Language Models} تقویت شده‌اند، قادر به بهبود عملکرد معاملاتی در بازارهای مالی هستند یا خیر.  
روش‌های یادگیری تقویتی در سال‌های اخیر به‌طور فزاینده‌ای در حوزه‌های معاملات الگوریتمی و مدیریت سبد سهام\footnote{Portfolio} به‌کار گرفته شده‌اند \cite{pricope2021deep} و ازاین‌رو بستری مناسب برای این پژوهش فراهم می‌آورند. 

برای تحقق این هدف، پژوهش حاضر اهداف زیر را دنبال می‌کند:

\begin{itemize}
    \item توسعه و ارزیابی عامل‌های یادگیری تقویتی برای معامله‌گری در بازارهای مالی.
\item یکپارچه‌سازی ویژگی‌های احساسی استخراج‌شده از اخبار مرتبط (با بهره‌گیری از مدل‌های زبانی بزرگ) همراه با داده‌های بازار و شاخص‌های فنی.

\item مقایسه عملکرد مدل‌های پیشنهادی با مدل‌های سنتی، راهبردهای مبنا\footnote{Baseline} و یافته‌های پژوهش‌های دانشگاهی اخیر.
\item بررسی تأثیر طراحی \textbf{فضای حالت}\footnote{State Space} و \textbf{فضای عمل}\footnote{Action Space}، \textbf{شکل‌دهی پاداش}\footnote{Reward Shaping} و انتخاب \textbf{ابرپارامترها}\footnote{Hyperparameters} بر کارایی مدل.
\item ارزیابی الگوریتم‌های پیشرفته و بررسی ادغام معماری‌هایی مانند \lr{LSTM} در این الگوریتم‌ها به‌منظور بهبود عملکرد معاملاتی.
\end{itemize}

دامنه پژوهش به‌صورت زیر تعریف می‌شود:

\begin{itemize}
    \item تحلیل صرفاً محدود به اجزای شاخص \lr{Dow Jones Industrial Average (Dow30)} است و به سایر طبقات دارایی یا بازارهای جهانی تعمیم داده نمی‌شود.
    \item داده‌های بازار شامل داده‌های \lr{OHLCV}\footnote{Open, High, Low, Close, Volume (OHLCV)} و سیگنال‌های احساسی مبتنی بر \lr{FinBERT} \cite{araci2019finbert}، \lr{FinGPT} \cite{yang2023fingpt} و اخبار گوگل است. منابع جایگزین مانند شبکه‌های اجتماعی یا داده دفتر سفارش\footnote{Order Book} در این پژوهش مدنظر قرار نمی‌گیرند.
    \item تمرکز پژوهش بر افق‌های معاملاتی روزانه است و معاملات با فرکانس بسیار بالا\footnote{High-Frequency Trading} خارج از دامنه بررسی قرار دارند.
    \item مدل‌ها صرفاً در محیط پژوهشی طراحی و ارزیابی می‌شوند و هدف، توسعه سامانه‌های معاملاتی آماده بهره‌برداری نیست.
\item ارزیابی‌ها بر اساس شاخص‌های رایج عملکرد و ریسک انجام می‌شود. بهینه‌سازی سبد سرمایه‌گذاری بلندمدت و راهبردهای تخصیص سرمایه در این پژوهش بررسی نمی‌شوند.
\end{itemize}




\section{کاربردها و اهمیت عملی}

توسعه \textbf{عامل‌های یادگیری تقویتی} تقویت‌شده با \textbf{مدل‌های زبانی بزرگ} دارای پتانسیل کاربردی چشمگیری در حوزه‌های معامله‌گری الگوریتمی و مدیریت سبد سرمایه‌گذاری است. با ترکیب داده‌های ساختاریافته بازار با اطلاعات احساسی غیرساختاریافته، این سامانه‌ها قادر خواهند بود تصمیم‌گیری \textbf{انطباقی و بسترآگاه}\footnote{Context-Aware} را پشتیبانی کرده و در نتیجه عملکرد تعدیل‌شده بر اساس ریسک را بهبود دهند.  
چارچوب پیشنهادی این پژوهش صرفاً به آزمایش‌های نظری محدود نیست، بلکه قابلیت به‌کارگیری مستقیم در خطوط لوله معاملاتی خودکار، سامانه‌های پشتیبان تصمیم‌گیری سرمایه‌گذاری و بسترهای پایش ریسک مالی را نیز دارد.

با وجود آنکه معامله مستقیم در بازارهای سهام بین‌المللی از ایران به دلیل محدودیت‌های نظارتی و ژئوپولیتیکی در حال حاضر امکان‌پذیر نیست، پیاده‌سازی عملی همچنان از طریق کارگزاران واسطه و پلتفرم‌های معاملاتی برخط (مانند \lr{RoboForex} و ارائه‌دهندگان مشابه) میسر است. این واسطه‌ها دسترسی به شاخص‌های جهانی سهام، بازار ارز و سایر طبقات دارایی را فراهم می‌کنند و در نتیجه ادغام عامل‌های مبتنی بر یادگیری تقویتی در سناریوهای معاملاتی واقعی را امکان‌پذیر می‌سازند. بنابراین، هرچند مدل‌های ارائه‌شده در این پژوهش رویکردی پژوهشی دارند، می‌توان آن‌ها را برای اجرای نیمه‌خودکار یا تمام‌خودکار از طریق اتصال به \textbf{رابط‌های برنامه‌نویسی کارگزاری}\footnote{Brokerage APIs} تطبیق و پیاده‌سازی کرد.

زمان‌بندی این پژوهش نیز اهمیت ویژه‌ای دارد. با پیشرفت سریع فناوری‌های یادگیری تقویتی و مدل‌های زبانی بزرگ، بازارهای مالی وارد مرحله‌ای تازه از تصمیم‌گیری مبتنی بر هوش مصنوعی شده‌اند. توانایی یادگیری تقویتی در بهینه‌سازی راهبردهای دنباله‌دار و قابلیت مدل‌های زبانی بزرگ در تفسیر حجم انبوه داده‌های متنی و احساسی، مزیتی هم‌افزا برای سرمایه‌گذاران ایجاد می‌کند. در عصری که با نوسان‌پذیری، جهانی‌شدن و اهمیت روزافزون معامله‌گری مبتنی بر اطلاعات تعریف می‌شود، سامانه‌های یکپارچه مبتنی بر یادگیری تقویتی و مدل‌های زبانی بزرگ می‌توانند منافع مالی قابل‌توجهی همچون بازدهی بالاتر و مدیریت ریسک مقاوم‌تر را فراهم آورند.

\section{پرسش‌های پژوهش و فرضیه‌ها}

این پژوهش بر اساس پرسش‌های زیر و فرضیه‌های متناظر با آن‌ها سازمان‌دهی شده است:

\subsection*{کارایی یادگیری تقویتی در مقایسه با رویکردهای سنتی }
\begin{itemize}
    \item \textbf{پرسش پژوهش ۱ (RQ1):} آیا \textbf{عامل‌های یادگیری تقویتی}\footnote{Reinforcement Learning (RL)} در مقایسه با مدل‌های سنتی، راهبردهای مبنا و رویکردهای گزارش‌شده در پژوهش‌های دیگر در معاملات شاخص \lr{Dow30} و سایر وظایف معاملاتی بازار عملکرد بهتری دارند؟ \\
    \textbf{فرضیه ۱ (H1):} انتظار می‌رود عامل‌های یادگیری تقویتی بازدهی تعدیل‌شده بر اساس ریسک بالاتری نسبت به مدل‌های سنتی و راهبردهای مبنا ارائه دهند؛ هرچند ممکن است در مقایسه با پیشرفته‌ترین رویکردهای روز همواره برتری نداشته باشند.
\end{itemize}

\subsection*{تأثیر داده‌های احساسی و داده‌های بازار}
\begin{itemize}
\item \textbf{پرسش پژوهش ۲ (RQ2):} آیا افزودن ویژگی‌های احساسی استخراج‌شده از اخبار مرتبط موجب بهبود عملکرد معاملاتی نسبت به مدل‌های صرفاً مبتنی بر قیمت می‌شود؟ \\
\textbf{فرضیه ۲ (H2):} انتظار می‌رود ویژگی‌های احساسی استخراج‌شده از اخبار مرتبط عملکرد معاملاتی را نسبت به مدل‌های صرفاً مبتنی بر داده‌های قیمتی بهبود دهند.
    
\item \textbf{پرسش پژوهش ۳ (RQ3):} آیا روند قیمت سهام اعتبار ویژگی‌های احساسی استخراج‌شده از اخبار را تأیید می‌کند؟ \\
\textbf{فرضیه ۳ (H3):} انتظار می‌رود ویژگی‌های احساسی دارای ارتباط با حرکات قیمتی باشند و بتوانند اطلاعات تکمیلی ارزشمندی برای بهبود عملکرد مدل فراهم کنند.
    
    \item \textbf{پرسش پژوهش ۴ (RQ4):} \textbf{شکل‌دهی پاداش} و طراحی \textbf{فضای عمل} و \textbf{فضای حالت} چه تأثیری بر عملکرد مدل‌های معاملاتی مبتنی بر یادگیری تقویتی دارند؟ \\
    \textbf{فرضیه ۴ (H4):} انتظار می‌رود شکل‌دهی پاداش و طراحی دقیق فضای عمل و فضای حالت تأثیر مثبتی بر عملکرد عامل‌های معاملاتی مبتنی بر یادگیری تقویتی داشته باشند.
\end{itemize}

\subsection*{معماری مدل‌ها و ابرپارامترها}
\begin{itemize}
    \item \textbf{پرسش پژوهش ۵ (RQ5):} آیا ترکیب \lr{LSTM}\footnote{Long Short-Term Memory (LSTM)} با الگوریتم‌های یادگیری تقویتی مانند \lr{Recurrent PPO (RPPO)} به بهبود پایدار عملکرد منجر می‌شود؟ \\
    \textbf{فرضیه ۵ (H5):} ترکیب معماری‌های \lr{LSTM} با الگوریتم‌های یادگیری تقویتی مانند \lr{RPPO} با افزایش توانایی مدل در تشخیص الگوهای دنباله‌ای، عملکرد آن را بهبود خواهد داد.
        
\item \textbf{پرسش پژوهش ۶ (RQ6):} کدام رویکرد استخراج و بازنمایی ویژگی‌های احساسی برای معامله‌گری مالی کارآمدتر است؟ \\
\textbf{فرضیه ۶ (H6):} انتظار می‌رود رویکردهای ترکیبی در استخراج احساسات، بازنمایی مؤثرتری نسبت به استفاده منفرد از یک روش ارائه دهند.
    
    \item \textbf{پرسش پژوهش ۷ (RQ7):} \textbf{ابرپارامترهای الگوریتمی}\footnote{Hyperparameters} تا چه اندازه بر عملکرد و پایداری راهبردهای معاملاتی مبتنی بر یادگیری تقویتی اثرگذارند؟ \\
    \textbf{فرضیه ۷ (H7):} انتظار می‌رود ابرپارامترهای الگوریتمی نقش مهم و تعیین‌کننده‌ای در بهبود یا تضعیف عملکرد و پایداری مدل‌های معاملاتی مبتنی بر یادگیری تقویتی داشته باشند.
\end{itemize}


\section{ساختار گزارش}

این گزارش در نه فصل تنظیم شده است:  

\textbf{فصل اول} مقدمه است و به اهداف پژوهش، پرسش‌ها و فرضیه‌ها، اهمیت عملی و دامنه مطالعه می‌پردازد.  

\textbf{فصل دوم} مبانی نظری پژوهش را معرفی می‌کند. این فصل به مرور اصول یادگیری تقویتی، مدل‌های زبانی بزرگ، داده‌ها و اطلاعات مرتیط به بازارهای مالی و محدودیت‌های عملی معامله‌گری می‌پردازد.

\textbf{فصل سوم} مروری بر ادبیات مرتبط ارائه می‌دهد. این فصل به بررسی پژوهش‌های مربوط به کاربرد مدل‌های زبانی بزرگ و یادگیری تقویتی در حوزه مالی و رویکردهای ترکیبی آن‌ها در معامله‌گری می‌پردازد.

\textbf{فصل چهارم} روش‌شناسی پژوهش و طراحی کلی سامانه را توضیح می‌دهد و به تعریف رسمی مسئله و معماری چارچوب پیشنهادی می‌پردازد.  

\textbf{فصل پنجم} به «توصیف پروژه» اختصاص دارد و شامل زیربخش‌های زیر است:  
\begin{itemize}
    \item \textbf{مقدمه}
    \item \textbf{توصیف بازار انتخاب‌شده}
    \item \textbf{گردآوری داده‌ها}
    \item \textbf{پیش‌پردازش داده‌ها}
    \item \textbf{تحلیل داده‌ها}
    \item \textbf{استخراج احساسات از اخبار}
    \item \textbf{محیط معامله‌گری}
    \item \textbf{عامل یادگیری تقویتی}
    \item \textbf{پایش}
    \item \textbf{ارائه و استقرار}
\end{itemize}

\textbf{فصل ششم} به پیاده‌سازی اختصاص دارد و جزئیات فرآیند اجرا، ابزارها و چارچوب‌های مورد استفاده، ملاحظات مربوط به بازتولیدپذیری و منابع محاسباتی را بیان می‌کند.  

\textbf{فصل هفتم} طراحی آزمایش‌ها و نتایج تجربی حاصل از ارزیابی سامانه پیشنهادی را تشریح می‌کند.  

\textbf{فصل هشتم} شامل بحث و تحلیل انتقادی یافته‌ها است و بر نقاط قوت، دستاوردها و محدودیت‌های پژوهش تأکید دارد.  

در نهایت، \textbf{فصل نهم} به جمع‌بندی مطالعه، بررسی پیامدهای آن برای حوزه \textbf{هوش مصنوعی مالی} و پیشنهاد مسیرهای بالقوه برای تحقیقات آینده اختصاص دارد.  


\chapter{مفاهیم نظری}

\section{مقدمه}

این فصل به مبانی نظری مورد نیاز برای درک توسعه یک \textbf{عامل یادگیری تقویتی} در بستر بازارهای مالی، با تأکید بر ادغام آن با \textbf{مدل‌های زبانی بزرگ}، می‌پردازد.  

ابتدا اصول پایه‌ای یادگیری تقویتی معرفی می‌شوند؛ از جمله اجزای اصلی آن، موازنه اکتشاف ـ بهره‌برداری\footnote{Exploration--Exploitation Balance}، و تمایز میان رویکردهای درون‌سیاستی\footnote{On-Policy Learning} و برون‌سیاستی\footnote{Off-Policy Learning}، همچنین مدل‌مبنا\footnote{Model-Based} در برابر مدل‌نا‌مبنا\footnote{Model-Free}. بر این اساس، الگوریتم‌های کلیدی یادگیری تقویتی از \lr{Q-learning} و روش‌های \textbf{گرادیان سیاست}\footnote{Policy Gradient Methods} تا چارچوب‌های \textbf{بازیگر ـ منتقد}\footnote{Actor--Critic Frameworks} و الگوریتم‌های پیشرفته‌ای مانند \lr{Proximal Policy Optimization (PPO)} و نسخه بازگشتی آن \lr{Recurrent PPO} مرور خواهند شد.  

در ادامه، مبانی \textbf{مدل‌های زبانی بزرگ} بررسی می‌شوند. این بخش شامل توضیح معماری ترنسفورمر\footnote{Transformer Architecture}، روش‌های تعبیه\footnote{Embedding Techniques} و پارادایم‌های مختلف آموزش است. همچنین نقش تحلیل احساسات در حوزه مالی برجسته می‌شود تا نشان دهد چگونه داده‌های متنی می‌توانند مکمل شاخص‌های عددی بازار باشند.  

در نهایت، ساختار و چالش‌های داده‌های بازار مالی، اصول مدیریت سبد سهام و محدودیت‌های عملی مانند هزینه‌های تراکنش، معیارهای ریسک و شاخص‌های ارزیابی تشریح می‌شوند. به‌طور کلی، این مفاهیم بنیان نظری لازم برای طراحی و ارزیابی یک عامل معاملاتی مبتنی بر یادگیری تقویتی را که با بینش‌های استخراج‌شده از مدل‌های زبانی بزرگ تقویت شده است، فراهم می‌آورند.


\section{مبانی یادگیری تقویتی}

\subsection{اجزای اصلی}

\textbf{یادگیری تقویتی} ~\cite{sutton2018reinforcement} محاسباتی برای تصمیم‌گیری دنباله‌ای است که در آن عامل با تعامل مستقیم با محیط می‌آموزد چگونه عمل‌های مناسب را انتخاب کند. هدف اصلی این رویکرد، بیشینه‌سازی پاداش تجمعی در طول زمان است. اجزای اصلی یادگیری تقویتی در شکل \ref{fig:rl_framework} نمایش داده شده‌اند.  

\begin{figure}[htbp]
  \centering
  \includegraphics[width=0.75\textwidth]{photos/reinforcement_learning_diagram.png}
  \caption{چارچوب کلی یادگیری تقویتی~\cite{mathworksRLdiag2025}.}
  \label{fig:rl_framework}
\end{figure}

\begin{itemize}
    \item \textbf{عامل:} یادگیرنده یا تصمیم‌گیرنده. در معامله‌گری الگوریتمی، عامل همان راهبرد معاملاتی است که عمل‌های خرید، فروش یا نگه‌داشتن را انتخاب می‌کند.
    
    \item \textbf{محیط:} سامانه خارجی که عامل با آن تعامل دارد. در کاربردهای مالی، محیط معمولاً بازار است که شامل پویایی قیمت، دفتر سفارش و داده‌های احساسی می‌شود.
    
    \item \textbf{حالت ($s_t$):} بازنمایی وضعیت محیط در زمان $t$. در معامله‌گری، حالت‌ها می‌توانند شامل شاخص‌های فنی  وامتیازهای احساسی استخراج‌شده از اخبار و داده‌های تاریخی باشند.
    
    \item \textbf{عمل ($a_t$):} مجموعه تصمیم‌های در دسترس عامل در زمان $t$. در ساده‌ترین حالت، این مجموعه شامل $\{ \text{خرید، فروش، نگه‌داشتن} \}$ است، اما می‌تواند به بازتخصیص سبد سهام یا تعیین اندازه موقعیت به‌صورت پیوسته نیز گسترش یابد.
    
    \item \textbf{پاداش ($r_t$):} سیگنال بازخورد عددی که سود یا زیان فوری ناشی از اجرای عمل $a_t$ در حالت $s_t$ را نشان می‌دهد. در زمینه مالی، این معمولاً به‌صورت \textbf{سود و زیان}، بازدهی تعدیل‌شده بر اساس ریسک یا تابعی سفارشی تعریف می‌شود که اهداف مدیریت ریسک را نیز در نظر می‌گیرد.
    
    \item \textbf{سیاست ($\pi$):} راهبرد عامل برای نگاشت حالت‌ها به عمل‌ها. سیاست‌ها می‌توانند \textbf{قطعی} باشند ($a = \pi(s)$) یا \textbf{تصادفی} باشند ($a \sim \pi(\cdot|s)$).
    
    \item \textbf{تابع ارزش:} برآورد پاداش تجمعی مورد انتظار هنگام شروع از یک حالت یا جفت حالت ـ عمل مشخص. این تابع معیاری از مطلوبیت بلندمدت حالت‌ها، فراتر از پاداش‌های فوری ارائه می‌دهد.
    
    \item \textbf{مدل محیط:} یک مؤلفه اختیاری که انتقال حالت‌ها را بر اساس عمل‌ها پیش‌بینی می‌کند. رویکردهای \textbf{مدل‌نا‌مبنا}\footnote{Model-Free} مانند \lr{Proximal Policy Optimization, PPO} محیط را به‌طور صریح مدل‌سازی نمی‌کنند، در حالی که روش‌های \textbf{مدل‌مبنا}\footnote{Model-Based} چنین مدل‌های پیش‌بینی را در نظر می‌گیرند.
\end{itemize}

این اجزا شالوده الگوریتم‌های یادگیری تقویتی را تشکیل می‌دهند. در معامله‌گری مالی، طراحی دقیق فضای حالت، مجموعه عمل و تابع پاداش اهمیت حیاتی دارد؛ زیرا تعریف‌های ضعیف یا نادرست می‌توانند به پویایی‌های یادگیری ناپایدار یا راهبردهای معاملاتی زیان‌ده منجر شوند.

\subsection{اکتشاف\protect\footnote{\lr{Exploration}} در برابر بهره‌برداری\protect\footnote{\lr{Exploitation}}}

یکی از چالش‌های اصلی در \textbf{یادگیری تقویتی} موازنه میان \textbf{اکتشاف} و \textbf{بهره‌برداری} است.  

\begin{itemize}
    \item \textbf{اکتشاف:} فرآیند انتخاب عمل‌هایی با قطعیت کمتر برای کسب دانش جدید و احتمالاً کشف راهبردهایی که بازدهی بلندمدت بیشتری ایجاد می‌کنند. در معامله‌گری، اکتشاف می‌تواند شامل آزمون شاخص‌ها یا سیگنال‌های جدید بازار باشد، حتی اگر سودآوری فوری آن‌ها نامشخص باشد.
    
    \item \textbf{بهره‌برداری:} فرآیند استفاده از دانش موجود برای بیشینه‌سازی پاداش مورد انتظار از طریق انتخاب مداوم عمل‌هایی که بهترین عملکرد برآوردشده را دارند. در کاربردهای مالی، این به معنای انجام معاملاتی است که در گذشته بازدهی قوی نشان داده‌اند.
\end{itemize}

  \centering
  \includegraphics[width=0.75\textwidth]{photos/exp_vs_exp.jpg}
  \caption{نمایی از موازنه اکتشاف ـ بهره‌برداری~\cite{huggingfaceExp2RL2025}.}

تعادل میان اکتشاف و بهره‌برداری حیاتی است. بهره‌برداری بیش از حد می‌تواند به همگرایی زودهنگام بر سیاست‌های زیربهینه منجر شود و عملکرد بلندمدت را محدود کند. در مقابل، اکتشاف بیش از حد ممکن است به رفتار ناپایدار و افزایش مواجهه با ریسک بیانجامد ـ موضوعی که در بازارهای مالی اهمیت ویژه‌ای دارد.  

\subsection{وظایف رویدادمحور\protect\footnote{\lr{Episodic Tasks}} در برابر وظایف پیوسته\protect\footnote{\lr{Continuous Tasks}}}

مسائل \textbf{یادگیری تقویتی} را می‌توان به‌طور کلی به \textbf{وظایف رویدادمحور} \textbf{وظایف پیوسته} تقسیم کرد که بر اساس نحوه تعامل عامل با محیط تعریف می‌شوند.

\begin{itemize}
    \item \textbf{وظایف رویدادمحور:} تعامل به دوره‌های گسسته تقسیم می‌شود که هرکدام حالت آغازین و پایانی مشخص دارند. در پایان هر اپیزود، محیط بازنشانی می‌شود و فرآیند یادگیری دوباره آغاز می‌گردد. نمونه‌های کلاسیک شامل بازی‌های تخته‌ای مانند شطرنج و \lr{Go} هستند که هر مسابقه یک اپیزود محسوب می‌شود. در معامله‌گری مالی، یک افق سرمایه‌گذاری ثابت (مثلاً شبیه‌سازی ۳۰ روز معاملاتی) را می‌توان به‌عنوان یک وظیفه اپیزودیک مدل کرد.
    
    \item \textbf{وظایف پیوسته:} تعامل به‌طور نامحدود ادامه دارد، بدون اینکه حالت پایانی طبیعی داشته باشد. عامل باید رفتار خود را در یک جریان بی‌پایان از حالت‌ها و پاداش‌ها بهینه کند. بازارهای مالی ذاتاً پیوسته‌اند، زیرا فرصت‌های معاملاتی و حرکت‌های قیمتی بدون نقطه پایان از پیش تعیین‌شده در جریان هستند.
\end{itemize}

این تمایز به‌طور مستقیم بر نحوه محاسبه توابع ارزش اثر می‌گذارد. در وظایف رویدادمحور معمولاً پاداش‌ها تا یک نقطه پایان مشخص جمع می‌شوند (بازدهی با افق محدود). اما در وظایف پیوسته از بازدهی تنزیل‌شده با افق نامحدود استفاده می‌شود تا مجموع پاداش‌ها همواره مقدار محدودی داشته باشد:
\[
G_t = \sum_{k=0}^{\infty} \gamma^k r_{t+k+1}, \quad 0 \leq \gamma < 1,
\]

که در آن $\gamma$ \textbf{ضریب تنزیل}\protect\footnote{\lr{Discount Factor}} است و موازنه میان پاداش‌های فوری و بلندمدت را کنترل می‌کند.

در معامله‌گری الگوریتمی، محیط‌ها غالباً به‌صورت پیوسته مدل می‌شوند تا ماهیت همیشگی بازار بازتاب یابد. با این حال، فرآیند آموزش معمولاً در قالب اپیزودها (مانند دوره‌های سالانه یا فصلی) سازمان‌دهی می‌شود تا یادگیری پایدارتر گردد، ارزیابی تسهیل شود و مقایسه میان راهبردها امکان‌پذیر باشد.  

\subsection{یادگیری درون‌سیاستی\protect\footnote{\lr{On-Policy Learning}} در برابر برون‌سیاستی\protect\footnote{\lr{Off-Policy Learning}}}

الگوریتم‌های \textbf{یادگیری تقویتی} را می‌توان به \textbf{روش‌های درون‌سیاستی} و \textbf{روش‌های برون‌سیاستی} تقسیم کرد. این دسته‌بندی بر اساس نحوه بهبود سیاست عامل تعریف می‌شود.  

\begin{itemize}
    \item \textbf{یادگیری درون‌سیاستی:} عامل ارزش سیاستی را می‌آموزد که هم‌اکنون در حال اجراست. داده‌ها با پیروی از همین سیاست جمع‌آوری می‌شوند و بهبودها بر همان اساس صورت می‌گیرد. نمونه‌ها شامل \lr{SARSA} و روش‌های \textbf{گرادیان سیاست}\footnote{Policy Gradient Methods} مانند \lr{Proximal Policy Optimization, PPO} هستند. در معامله‌گری الگوریتمی، این به عاملی اشاره دارد که بر اساس معاملاتی که تحت راهبرد جاری خود انجام می‌دهد، تطبیق پیدا می‌کند.
    
    \item \textbf{یادگیری برون‌سیاستی:} عامل ارزش یک سیاست بهینه را می‌آموزد، حتی اگر از یک سیاست رفتاری متفاوت پیروی کند. این رویکرد امکان یادگیری از داده‌های تاریخی، نمایش‌ها یا رفتارهای اکتشافی را فراهم می‌آورد. نمونه‌های بارز شامل \lr{Q-learning} و \lr{Deep Q-Networks, DQN} هستند. در معامله‌گری، این روش به عامل اجازه می‌دهد از داده‌های تاریخی گسترده بازار بیاموزد، بدون آنکه معاملات را در زمان واقعی اجرا کند.
\end{itemize}

روش‌های درون‌سیاستی معمولاً پایدارتر و ایمن‌تر در حوزه‌های پرریسک مانند معامله‌گری هستند، در حالی که روش‌های برون‌سیاستی داده‌محورتر بوده و می‌توانند از تجربیات گذشته و شبیه‌سازی‌های برون‌خط استفاده کنند~\cite{sutton2018reinforcement}.  


\subsection{روش‌های مدل‌مبنا\protect\footnote{\lr{Model-Based Methods}} در برابر مدل‌نامبنا\protect\footnote{\lr{Model-Free Methods}}}

تمایز مهم دیگر در یادگیری تقویتی میان \textbf{روش‌های مدل‌نا‌مبنا} و \textbf{روش‌های مدل‌مبنا} است. این تمایز بر اساس صریح‌بودن یا نبودن مدل‌سازی محیط تعریف می‌شود. 
 
\begin{itemize}
    \item \textbf{روش‌های بدون‌مدل:} این الگوریتم‌ها تلاشی برای یادگیری ساختار انتقال محیط نمی‌کنند، بلکه مستقیماً توابع ارزش یا سیاست‌ها را از طریق تعامل تخمین می‌زنند. نمونه‌هایی از این دسته شامل \lr{Q-learning}، \lr{DQN} و \lr{PPO} هستند. در معامله‌گری، یک عامل بدون‌مدل مستقیماً از داده‌های بازار مانند سری‌های قیمتی، شاخص‌ها و سیگنال‌های احساسی یاد می‌گیرد، بدون آنکه مسیر آینده بازار را پیش‌بینی کند.
    
    \item \textbf{روش‌های مبتنی‌بر‌مدل:} این رویکردها بر یادگیری یا استفاده از یک مدل پیش‌بینی‌کننده رفتار محیط (مانند احتمال‌های گذار بین حالت‌ها و تابع پاداش) تکیه دارند. چنین مدل‌هایی امکان به‌کارگیری تکنیک‌های برنامه‌ریزی مانند \textbf{برنامه‌ریزی پویا}\footnote{Dynamic Programming} یا \textbf{جستجوی درختی مونت‌کارلو}\footnote{Monte Carlo Tree Search, MCTS} را فراهم می‌کنند. در حوزه مالی، این می‌تواند شامل ساخت مدل‌های پیش‌بینی تغییرات قیمت دارایی‌ها یا شبیه‌سازی رفتار دفتر سفارش برای هدایت تصمیم‌گیری باشد.
\end{itemize}

به طور کلی، روش‌های بدون‌مدل ساده‌تر بوده و با یادگیری عمیق بهتر مقیاس‌پذیر هستند، هرچند به داده و فرآیند اکتشاف بیشتری نیاز دارند. در مقابل، روش‌های مبتنی‌بر‌مدل از نظر استفاده از داده کاراترند، اما نسبت به خطاهای مدل حساسیت بیشتری دارند ـ موضوعی که در بازارهای مالی پرتلاطم و ناپایدار اهمیت ویژه‌ای پیدا می‌کند.

\subsection{Q-Learning و شبکه‌های Q عمیق\protect\footnotemark}
\footnotetext{\lr{Deep Q-Networks, DQN}}

\textbf{Q-Learning} که نخستین‌بار توسط \lr{Watkins} در سال 1989 معرفی شد~\cite{watkins1989q}، یکی از الگوریتم‌های بنیادی در \textbf{یادگیری تقویتی} است. این روش \textbf{برون‌سیاستی} و \textbf{مدل‌نا‌مبنا} است و هدف آن یادگیری تابع ارزش عمل بهینه $Q^*(s, a)$ می‌باشد. این تابع بیانگر پاداش تجمعی تنزیل‌شده مورد انتظار از انتخاب عمل $a$ در حالت $s$ و سپس پیروی از سیاست بهینه است. قاعده به‌روزرسانی Q-learning به‌صورت زیر تعریف می‌شود:

\[
Q(s_t, a_t) \leftarrow Q(s_t, a_t) + \alpha \Big[ r_{t+1} + \gamma \max_{a'} Q(s_{t+1}, a') - Q(s_t, a_t) \Big]
\]

که در آن $\alpha$ \textbf{نرخ یادگیری}\footnote{Learning Rate}، $\gamma$ \textbf{ضریب تنزیل}\footnote{Discount Factor} و $r_{t+1}$ پاداش دریافتی پس از اجرای عمل $a_t$ در حالت $s_t$ است.  

\bigskip

\textbf{شبکه‌های Q عمیق}\footnote{Deep Q-Networks, DQN} بسط Q-learning هستند که در آن تابع $Q$ با استفاده از یک شبکه عصبی عمیق تقریب زده می‌شود. این رویکرد امکان مقیاس‌پذیری به فضاهای ورودی با ابعاد بالا مانند تصاویر یا سری‌های زمانی مالی را فراهم می‌کند. پژوهش برجسته \lr{Mnih et al. (2015)}~\cite{mnih2015human} موفقیت DQN را در یادگیری مستقیم از ورودی‌های خام پیکسلی در بازی‌های آتاری نشان داد. نوآوری‌های کلیدی DQN شامل موارد زیر است:  

\begin{itemize}
\item \textbf{بازپخش تجربه}\footnote{Experience Replay}: انتقال‌ها $(s, a, r, s')$ در یک حافظه بازپخش ذخیره می‌شوند و آموزش شبکه با استفاده از \textbf{بسته‌های کوچک داده} که به‌طور تصادفی از این حافظه نمونه‌برداری می‌شوند انجام می‌شود. این فرآیند همبستگی میان به‌روزرسانی‌های متوالی را کاهش داده و پایداری آموزش را افزایش می‌دهد.

    \item \textbf{شبکه‌های هدف}: یک شبکه هدف جداگانه برای محاسبه مقادیر هدف در Q-learning به‌کار گرفته می‌شود که به‌صورت دوره‌ای به‌روزرسانی می‌گردد. این رویکرد از نوسانات مخرب و واگرایی در طول آموزش جلوگیری می‌کند.
\end{itemize}

\bigskip

در کاربردهای معاملاتی، Q-learning و DQN می‌توانند برای یادگیری سیاست‌هایی مورد استفاده قرار گیرند که مشخص می‌کنند آیا عامل باید دارایی را بخرد، بفروشد یا نگه دارد؛ بر اساس حالتی که توسط شاخص‌های بازار، امتیازهای احساسی و داده‌های قیمتی تعریف می‌شود. با این حال، بازارهای مالی چالش‌های خاصی برای DQN ایجاد می‌کنند، زیرا حالت‌ها پیوسته، غیرایستا و پاداش‌ها تصادفی هستند؛ بنابراین تصمیم‌گیری حساس به ریسک ضرورت می‌یابد. با وجود این محدودیت‌ها، DQN همچنان به‌عنوان یک معیار مرجع ارزشمند در مقایسه با روش‌های پیشرفته‌تر مانند \textbf{گرادیان سیاست} و رویکردهای \textbf{بازیگر ـ منتقد} به‌کار می‌رود.  


\subsection{روش‌های گرادیان سیاست\protect\footnotemark}
\footnotetext{\lr{Policy Gradient Methods}}

در حالی که روش‌های مبتنی بر ارزش مانند Q-learning به تخمین \textbf{تابع ارزش عمل} می‌پردازند، \textbf{روش‌های گرادیان سیاست}\footnote{Policy Gradient Methods} مستقیماً سیاست $\pi_{\theta}(a|s)$ را که با پارامترهای $\theta$ (مانند وزن‌های یک شبکه عصبی) مشخص می‌شود، بهینه می‌کنند. هدف اصلی، بیشینه‌سازی بازدهی مورد انتظار است:

\[
J(\theta) = \mathbb{E}_{\pi_\theta}\bigg[ \sum_{t=0}^\infty \gamma^t r_t \bigg],
\]

که گرادیان‌ها بر اساس \textbf{قضیه گرادیان سیاست}\footnote{Policy Gradient Theorem} به‌دست می‌آیند~\cite{sutton1999policy}:

\[
\nabla_\theta J(\theta) = \mathbb{E}_{\pi_\theta}\big[ \nabla_\theta \log \pi_\theta(a_t|s_t) \, G_t \big],
\]

که در آن $G_t$ بازدهی پس از گام زمانی $t$ را نشان می‌دهد.  

\bigskip

\textbf{مزایای روش‌های گرادیان سیاست:}
\begin{itemize}
    \item قابلیت طبیعی در مدیریت فضاهای عمل \textbf{پیوسته} و با ابعاد بالا، در حالی که روش‌های مبتنی بر ارزش معمولاً به عمل‌های گسسته محدودند.
    \item پشتیبانی از سیاست‌های \textbf{تصادفی} که فرآیند اکتشاف را تسهیل می‌کنند.
    \item امکان ادغام مستقیم محدودیت‌ها، معیارهای ریسک یا هزینه‌های تراکنش در طراحی پاداش.
\end{itemize}

\textbf{چالش‌ها:}
\begin{itemize}
\item برآورد گرادیان در این روش‌ها معمولاً نوسان زیادی دارد (\textbf{واریانس بالا}\footnote{High Variance})، به همین دلیل لازم است از \textbf{روش‌های کاهش واریانس}\footnote{Variance Reduction Techniques} مانند استفاده از \textbf{مبناها}\footnote{Baselines} یا \textbf{توابع برتری}\footnote{Advantage Functions} کمک گرفته شود تا پایداری آموزش افزایش یابد.  

\item این الگوریتم‌ها در مقایسه با روش‌های \textbf{برون‌سیاستی} مانند \lr{DQN}، به داده‌های بیشتری برای یادگیری نیاز دارند و از نظر کارایی استفاده از نمونه‌ها ضعیف‌تر عمل می‌کنند.
\end{itemize}

\bigskip

در معامله‌گری مالی، روش‌های گرادیان سیاست به‌ویژه اهمیت دارند زیرا عمل‌ها (مانند وزن‌های سبد سهام یا اندازه موقعیت‌ها) ذاتاً پیوسته‌اند. برای نمونه، یک عامل یادگیری تقویتی می‌تواند به‌طور مستقیم کسری از سرمایه را که باید میان دارایی‌های مختلف تخصیص یابد، خروجی دهد. الگوریتم‌های اولیه‌ای مانند \lr{REINFORCE}~\cite{williams1992simple} بنیان این حوزه را شکل دادند، در حالی که بسط‌های مدرن مانند روش‌های بازیگر-منتقد پایداری و کارایی بیشتری به همراه داشته‌اند.  

\subsection{روش‌های بازیگر ـ منتقد\protect\footnotemark}
\footnotetext{\lr{Actor–Critic Methods}}

روش‌های \textbf{بازیگر ـ منتقد} مزایای رویکردهای مبتنی بر ارزش و مبتنی بر سیاست را در \textbf{یادگیری تقویتی} ترکیب می‌کنند. این روش‌ها از دو مؤلفه اصلی تشکیل شده‌اند~\cite{konda2000actor}:  

\begin{itemize}
    \item \textbf{بازیگر (Actor):} تابع سیاست $\pi_{\theta}(a|s)$ با پارامترهای $\theta$ که عمل‌ها را بر اساس حالت‌ها انتخاب می‌کند.
    \item \textbf{منتقد (Critic):} تابع ارزش $V^{\pi}(s)$ یا تابع ارزش عمل $Q^{\pi}(s,a)$ با پارامترهای $\phi$ که کیفیت سیاست را از طریق برآورد بازدهی مورد انتظار ارزیابی می‌کند.
\end{itemize}

منتقد در این چارچوب نقشی مانند یک \textbf{مبنا}\footnote{Baseline} دارد و کمک می‌کند تا نوسان زیاد (واریانس) در برآورد گرادیان سیاست کاهش یابد. بازیگر وظیفه دارد پارامترهای سیاست را در جهتی که \textbf{قضیه گرادیان سیاست}\footnote{Policy Gradient Theorem} مشخص کرده به‌روزرسانی کند. در مقابل، منتقد با استفاده از \textbf{یادگیری تفاضل زمانی}\footnote{Temporal-Difference, TD Learning} مقدارهای ارزش را بهبود می‌دهد و بازخورد دقیق‌تری به بازیگر می‌دهد.  

به‌طور خلاصه، به‌روزرسانی رایج بازیگر به صورت زیر نوشته می‌شود:

\[
\nabla_\theta J(\theta) \approx \mathbb{E}_{\pi_\theta}\big[ \nabla_\theta \log \pi_\theta(a_t|s_t) \, A^{\pi}(s_t, a_t) \big],
\]

که در آن $A^{\pi}(s_t, a_t)$ \textbf{تابع برتری}\footnote{Advantage Function} است. این تابع نشان می‌دهد که یک عمل در وضعیت خاص، نسبت به میانگین انتظاری (مبنا) تا چه حد بهتر یا بدتر است. به این ترتیب، بازیگر به سمت اعمالی متمایل می‌شود که ارزش بیشتری نسبت به مبنا دارند.

\bigskip

\textbf{مزایای روش‌های بازیگر-منتقد}
\begin{itemize}
    \item کارایی نمونه‌ای بیشتر نسبت به روش‌های صرفاً گرادیان سیاست مانند \lr{REINFORCE}، به دلیل کاهش واریانس توسط منتقد.
    \item مقیاس‌پذیری به فضاهای عمل پیوسته و با ابعاد بزرگ.
    \item ایجاد بنیان برای الگوریتم‌های پیشرفته‌ای مانند \lr{Advantage Actor–Critic, A2C}، \lr{Asynchronous Advantage Actor–Critic, A3C} و \lr{Proximal Policy Optimization, PPO}.
\end{itemize}

\bigskip

روش‌های Actor–Critic به‌ویژه در معامله‌گری مالی کارآمد هستند؛ زیرا منتقد با ارائه برآوردهای دقیق‌تر از بازدهی، یادگیری را در بازارهای پرنوسان و غیرایستا پایدارتر می‌سازد. برای نمونه، بازیگر می‌تواند معاملاتی مانند تنظیم اندازه موقعیت‌ها را پیشنهاد دهد، در حالی که منتقد سودآوری بلندمدت آن‌ها را ارزیابی می‌کند. بسط‌هایی مانند A3C نیز امکان آموزش موازی در چندین بازار شبیه‌سازی‌شده را فراهم کرده و کارایی و پایداری فرآیند یادگیری را افزایش می‌دهند.  


\subsection{\lr{Proximal Policy Optimization (PPO)}}

الگوریتم \lr{Proximal Policy Optimization (PPO)}~\cite{schulman2017proximal} یکی از پرکاربردترین روش‌های \textbf{گرادیان سیاست}است که به دلیل پایداری، سادگی و عملکرد تجربی قوی شناخته می‌شود. این الگوریتم در خانواده روش‌های \textbf{بازیگر ـ منتقد} قرار می‌گیرد، اما سازوکاری برای محدود کردن به‌روزرسانی‌های بیش از حد سیاست معرفی می‌کند؛ زیرا چنین به‌روزرسانی‌هایی می‌توانند آموزش را ناپایدار سازند.

ایده اصلی، بهینه‌سازی یک \textbf{تابع هدف جانشین} با استفاده از \textbf{نسبت احتمالات برش‌خورده}\footnote{Clipped Probability Ratio} است:  

\[
L^{\text{CLIP}}(\theta) = \mathbb{E}_t \Big[ 
    \min\big( r_t(\theta) A_t,\,
    \text{clip}(r_t(\theta), 1-\epsilon, 1+\epsilon) A_t \big)
\Big],
\]

که در آن:
\begin{itemize}
    \item $r_t(\theta) = \tfrac{\pi_\theta(a_t|s_t)}{\pi_{\theta_\text{old}}(a_t|s_t)}$ نسبت احتمالات است،
    \item $A_t$ \textbf{تابع برتری} برآوردشده است،
    \item $\epsilon$ پارامتر برش است (معمولاً در بازه $0.1$ تا $0.2$).
\end{itemize}

برش‌زدن مانع می‌شود که سیاست جدید در یک به‌روزرسانی بیش از حد از سیاست قدیم منحرف شود و به این ترتیب یادگیری پایدارتر می‌گردد. افزون بر این، PPO اغلب از \textbf{منظم‌سازی آنتروپی}\footnote{Entropy Regularization} برای تقویت اکتشاف بهره می‌گیرد.  

\bigskip
\textbf{تابع زیان کلی:}  
در عمل، PPO چندین مؤلفه را در یک تابع زیان ترکیب می‌کند:  

\[
L^{\text{PPO}}(\theta) = \mathbb{E}_t \Big[ 
    L^{\text{CLIP}}(\theta) 
    - c_1 \big( V_\theta(s_t) - V_t^{\text{target}} \big)^2 
    + c_2 H\big( \pi_\theta(\cdot|s_t) \big)
\Big],
\]

که در آن:  
\begin{itemize}
    \item $L^{\text{CLIP}}(\theta)$ زیان سیاست برش‌خورده است،
    \item $\big( V_\theta(s_t) - V_t^{\text{target}} \big)^2$ خطای مربعی میان برآورد منتقد و بازدهی هدف است،
    \item $H(\pi_\theta(\cdot|s_t))$ آنتروپی سیاست است که اکتشاف را تشویق می‌کند،
    \item $c_1$ و $c_2$ ضرایب کنترل‌کننده اهمیت مؤلفه‌های ارزش و آنتروپی هستند.
\end{itemize}

\textbf{مزایای PPO:}
\begin{itemize}
    \item موازنه مؤثر میان اکتشاف و بهره‌برداری،  
    \item ساده‌تر و مقاوم‌تر نسبت به روش‌های پیشین مبتنی بر ناحیه اعتماد مانند \textbf{TRPO}\footnote{Trust Region Policy Optimization, TRPO}،  
    \item مقیاس‌پذیری مناسب در ترکیب با شبکه‌های عصبی عمیق و مسائل با ابعاد بالا.  
\end{itemize}

PPO به‌ویژه برای محیط‌های معاملاتی مالی مناسب است؛ جایی که پایداری و مقاوم‌بودن اهمیت بالایی دارند زیرا داده‌ها پرنویز و غیرایستا هستند. با کنترل دقیق به‌روزرسانی‌های سیاست و ترکیب مؤلفه‌های سیاست، ارزش و آنتروپی، PPO امکان سازگاری تدریجی راهبردهای معاملاتی را فراهم می‌کند و میان بیشینه‌سازی سود، مدیریت ریسک و اکتشاف فرصت‌های جدید تعادل برقرار می‌سازد.  
  

\subsection{PPO بازگشتی\protect\footnotemark}
\footnotetext{\lr{Recurrent PPO}}

بسیاری از محیط‌های دنیای واقعی ــ از جمله بازارهای مالی ــ دارای \textbf{قابلیت مشاهده ناقص}\footnote{Partial Observability} هستند؛ به این معنا که عامل مستقیماً حالت زیربنایی را مشاهده نمی‌کند، بلکه تنها سیگنال‌های نویزی یا ناقصی مانند قیمت‌ها، شاخص‌های احساسی یا ویژگی‌های فنی دریافت می‌کند. برای رفع این چالش، الگوریتم \textbf{PPO} را می‌توان با \textbf{شبکه‌های عصبی بازگشتی}\footnote{Recurrent Neural Networks, RNNs} بسط داد که منجر به \textbf{PPO بازگشتی}\footnote{Recurrent PPO, RPPO} می‌شود~\cite{heess2015memory}.  

در RPPO، شبکه‌های سیاست و ارزش با مؤلفه‌های حافظه‌ای مانند \lr{LSTM} یا \lr{GRU} تقویت می‌شوند. این ساختار به عامل امکان می‌دهد \textbf{وابستگی‌های زمانی}\footnote{Temporal Dependencies} را ثبت کرده و یک \textbf{حالت پنهان}\footnote{Hidden State} را در طول گام‌های زمانی حفظ کند. حالت پنهان اطلاعات گذشته مرتبط را خلاصه می‌کند و در نتیجه تصمیم‌گیری مؤثرتری را در شرایط مشاهده ناقص ممکن می‌سازد.  

\textbf{مزایای PPO بازگشتی:}
\begin{itemize}
    \item توانایی مدیریت مشاهده ناقص از طریق حفظ حافظه شرایط گذشته بازار.  
    \item بهبود عملکرد در وظایف دنباله‌ای با وابستگی‌های بلندمدت.  
    \item ارائه چارچوبی واقع‌بینانه‌تر برای معامله‌گری مالی، جایی که تصمیم‌ها هم به بستر تاریخی و هم به سیگنال‌های جاری وابسته‌اند.  
\end{itemize}

الگوریتم \lr{Recurrent PPO (RPPO)} توانایی عامل معاملاتی مبتنی بر یادگیری تقویتی را در تشخیص الگوهای زمانی موجود در داده‌های مالی افزایش می‌دهد. برای نمونه، گاهی قیمت‌ها تمایل دارند مدتی در همان جهت قبلی حرکت کنند (اثر تداوم روند)، یا دوره‌های پرنوسان و کم‌نوسان بازار معمولاً پشت سر هم رخ می‌دهند. با ترکیب شاخص‌های فنی و ویژگی‌های احساسی در یک ساختار بازگشتی، \lr{RPPO} می‌تواند این الگوهای زمانی را بهتر از نسخه استاندارد \lr{PPO} (بدون حافظه بازگشتی) ثبت کرده و راهبردهای معاملاتی سازگارتر ارائه دهد.


\section{مبانی مدل‌های زبانی بزرگ}

\subsection{معماری Transformer}

\lr{Transformer}~\cite{vaswani2017attention} یک معماری یادگیری عمیق است که بنیان مدل‌های زبانی بزرگ مدرن را تشکیل می‌دهد. برخلاف \textbf{شبکه‌های عصبی بازگشتی} یا \textbf{مدل‌های کانولوشنی}\footnote{\lr{Convolutional Models}}، \lr{Transformer} کاملاً بر \textbf{سازوکار خودتوجهی}\footnote{\lr{Self-Attention Mechanism}} متکی است تا وابستگی میان واحدهای زبانی در متن را ثبت کند. این ویژگی امکان موازی‌سازی کارآمد و مقیاس‌پذیری تا مجموعه‌داده‌های بسیار بزرگ را فراهم می‌سازد.

\bigskip
\textbf{اجزای کلیدی معماری Transformer:}

\begin{itemize}
    \item \textbf{تعبیه ورودی}\footnote{\lr{Input Embeddings}}: هر \lr{Token} (واحد زبانی) در دنباله به یک بردار متراکم نگاشت می‌شود. از آنجا که \lr{Transformer}ها فاقد بازگشت یا کانولوشن هستند، \textbf{کدگذاری‌های مکانی}\footnote{\lr{Positional Encodings}} به این بردارها افزوده می‌شوند تا ترتیب \lr{Token}ها حفظ شود.
    
    \item \textbf{سازوکار خودتوجهی}: عملیات اصلی روابط میان همه جفت \lr{Token}‌ها در یک دنباله را محاسبه می‌کند. برای هر \lr{Token}، بردارهای پرسش ($Q$)، کلید ($K$) و مقدار ($V$) از نگاشت خطی تعبیه‌های ورودی به دست می‌آیند. توجه به صورت زیر محاسبه می‌شود:
    \[
    \text{Attention}(Q,K,V) = \text{softmax}\left(\frac{QK^\top}{\sqrt{d_k}}\right)V,
    \]
    که در آن $d_k$ بُعد بردارهای کلید است. این سازوکار به هر \lr{Token} اجازه می‌دهد بر اساس ارتباط معنایی به سایر \lr{Token}‌ها توجه کند.
    
    \item \textbf{توجه چندسری}\footnote{\lr{Multi-Head Attention}}: به‌جای یک عملیات توجه منفرد، چندین توجه به‌طور موازی محاسبه می‌شوند که امکان ثبت انواع مختلفی از روابط را فراهم می‌کند. خروجی‌ها سپس به هم متصل شده و به فضای تعبیه بازافکنی می‌شوند.
    
    \item \textbf{شبکه‌های پیشخور}\footnote{\lr{Feed-Forward Networks}}: بازنمایی هر موقعیت از یک شبکه عصبی تمام‌متصل با فعال‌ساز غیرخطی (مانند ReLU یا GELU) عبور می‌کند.
    
    \item \textbf{اتصالات باقیمانده و نرمال‌سازی لایه}\footnote{\lr{Residual Connections \& Layer Normalization}}: برای پایدارسازی آموزش و بهبود جریان گرادیان، اتصالات باقیمانده پیرامون لایه‌های توجه و پیشخور اعمال می‌شوند و سپس نرمال‌سازی لایه انجام می‌شود.
    
\item \textbf{ساختار رمزگذار ـ رمزگشا}\footnote{\lr{Encoder–Decoder Structure}}: \lr{Transformer} اصلی شامل یک رمزگذار (مفید برای بازنمایی ورودی‌ها مانند ترجمه) و یک رمزگشا (مسئول تولید دنباله) است. مدل‌های زبانی بزرگی مانند \lr{GPT} از گونه \textbf{فقط رمزگشا}\footnote{\lr{Decoder-Only}} با سازوکار \lr{Masked Self-Attention} استفاده می‌کنند تا اطمینان یابند پیش‌بینی در گام $t$ تنها به \lr{Token}‌های مشاهده‌شده تا آن لحظه وابسته باشد.

\end{itemize}


\subsection{جاسازی واژه‌ها و جاسازی‌های متنی}

مدل‌های \textbf{پردازش زبان طبیعی}\footnote{\lr{Natural Language Processing, NLP}} برای عملکرد مناسب نیازمند بازنمایی عددی از واژه‌ها هستند تا معنای واژگانی در آن‌ها منعکس شود. این هدف با استفاده از \textbf{جاسازی‌ها}\footnote{\lr{Embeddings}} محقق می‌شود؛ یعنی نگاشت توکن‌های گسسته به فضاهای برداری پیوسته که در آن‌ها روابط معنایی قابل اندازه‌گیری هستند.  

\bigskip
\textbf{جاسازی‌های ایستا}\footnote{\lr{Static Embeddings}}:  
روش‌های اولیه مانند \lr{Word2Vec}~\cite{mikolov2013efficient} و \lr{GloVe}~\cite{pennington2014glove} به هر واژه یک بردار ثابت اختصاص می‌دهند، مستقل از بافت آن. این جاسازی‌ها شباهت معنایی را از طریق حساب برداری ثبت می‌کنند (برای نمونه: $\text{king} - \text{man} + \text{woman} \approx \text{queen}$). با این حال، جاسازی‌های ایستا قادر به ثبت \textbf{چندمعنایی}\footnote{\lr{Polysemy}} نیستند؛ یعنی واژه‌هایی با معانی متعدد (مانند «bank») صرف‌نظر از بافت تنها با یک بردار نمایش داده می‌شوند.  

\bigskip
\textbf{جاسازی‌های متنی}\footnote{\lr{Contextual Embeddings}}:  
مدل‌های مدرن مبتنی بر ترنسفورمر مانند \lr{BERT}~\cite{devlin2019bert} و \lr{GPT}~\cite{radford2019language} جاسازی‌هایی تولید می‌کنند که \textbf{بافت‌محور} هستند. بازنمایی یک واژه بر اساس توکن‌های پیرامونی تغییر می‌کند و مدل را قادر می‌سازد میان معانی مختلف یک واژه تمایز قائل شود. این نوع جاسازی‌ها از سازوکار \textbf{خودتوجهی}\footnote{\lr{Self-Attention Mechanism}} برای ادغام اطلاعات معنایی در سراسر دنباله ورودی استفاده می‌کنند.  

به‌صورت رسمی، اگر $x_t$ جاسازی اولیه توکن $t$ باشد، جاسازی متنی $h_t$ به شکل زیر تعریف می‌شود:
\[
h_t = f_{\theta}(x_t \,|\, x_1, x_2, \ldots, x_T),
\]
که در آن $f_{\theta}$ تابع ترنسفورمر است که بازنمایی هر توکن را بر اساس کل دنباله شرطی می‌کند.  

\bigskip
\textbf{کاربرد در تحلیل احساسات مالی:}  
\begin{itemize}
    \item جاسازی‌های ایستا می‌توانند شباهت‌های معنایی کلی در اصطلاحات مالی را ثبت کنند (برای نمونه، رابطه میان «stock»، «equity» و «share»).  
    \item جاسازی‌های متنی امکان رفع ابهام از اصطلاحات خاص مالی را فراهم می‌کنند (برای نمونه، تمایز میان «bear» به معنای حیوان و «bear» به معنای احساس منفی بازار).  
    \item با بهره‌گیری از جاسازی‌های متنی، عامل‌های معاملاتی می‌توانند سیگنال‌های احساسی دقیق‌تری را از اخبار، تحلیل‌های کارشناسان و پست‌های شبکه‌های اجتماعی استخراج کنند و آن‌ها را در مدل‌های یادگیری تقویتی برای بهبود تصمیم‌های معاملاتی ادغام نمایند.  
\end{itemize}


\subsection{پیش‌آموزش در برابر ریزتنظیم و تنظیم مبتنی بر دستور}

\textbf{مدل‌های زبانی بزرگ}\footnote{\lr{Large Language Models, LLMs}} در چندین مرحله آموزش داده می‌شوند تا به‌تدریج از درک عمومی زبان به رفتار ویژه‌وظیفه‌ای یا همسو با کاربر سازگار شوند. سه پارادایم اصلی عبارت‌اند از:  

\bigskip
\textbf{پیش‌آموزش}\footnote{\lr{Pre-training}}:  
در نخستین مرحله، مدل بر روی متون عظیم به‌صورت خودنظارتی آموزش می‌بیند. اهداف رایج شامل \textbf{پیش‌بینی توکن بعدی}\footnote{\lr{Causal Language Modeling}} (مانند \lr{GPT}~\cite{radford2019language}) یا \textbf{پیش‌بینی توکن‌های ماسک‌شده}\footnote{\lr{Masked Language Modeling}} (مانند \lr{BERT}~\cite{devlin2019bert}) هستند. پیش‌آموزش، مدل را به دانش زبانی گسترده، دستور زبان و درک معنایی مجهز می‌سازد:
\[
\min_\theta \; \mathbb{E}_{x \sim D} \Big[ - \log p_\theta(x) \Big],
\]
که در آن $D$ مجموعه‌داده عظیم و $p_\theta$ توزیع مدل بر دنباله‌هاست.  

\bigskip
\textbf{ریزتنظیم}\footnote{\lr{Fine-tuning}}:  
پس از پیش‌آموزش، مدل می‌تواند با ادامه آموزش روی یک مجموعه‌داده برچسب‌خورده کوچک‌تر، به وظیفه‌ای خاص تطبیق یابد. برای نمونه، ریزتنظیم \lr{BERT} روی مجموعه‌داده‌ای با برچسب احساسات، تخصص در طبقه‌بندی احساسات را ممکن می‌سازد. در حوزه مالی، ریزتنظیم می‌تواند بر روی متون تخصصی مانند گزارش‌های تحلیلی، اخبار مالی یا داده‌های شبکه‌های اجتماعی انجام شود.  

\bigskip
\textbf{تنظیم مبتنی بر دستور}\footnote{\lr{Instruction Tuning}}:  
این رویکرد~\cite{ouyang2022training} پارادایمی جدیدتر است که در آن مدل‌ها بر روی مجموعه‌داده‌هایی شامل دستورهای زبان طبیعی همراه با پاسخ‌های مورد انتظار آموزش می‌بینند. این فرآیند مدل‌ها را به نیت انسانی نزدیک‌تر کرده و توانایی تعمیم در حالت \textbf{بدون نمونه}\footnote{\lr{Zero-shot}} و \textbf{کم‌نمونه}\footnote{\lr{Few-shot}} را بهبود می‌بخشد. مدل‌های تنظیم‌شده با دستور (مانند \lr{InstructGPT} و \lr{ChatGPT}) می‌توانند بدون نیاز به ریزتنظیم گسترده ویژه وظیفه، مستقیماً از طریق توصیف وظیفه عمل کنند.  

\bigskip
مراحل مختلف آموزش LLMها به‌طور مکمل در کاربردهای مالی نقش دارند:  
\begin{itemize}
    \item \textbf{پیش‌آموزش} دانش زبانی گسترده‌ای فراهم می‌کند که شناسایی احساسات عمومی و مفاهیم مالی در منابع متنی متنوع را ممکن می‌سازد.  
    \item \textbf{ریزتنظیم} روی متون مالی موجب تخصص مدل در واژگان حوزه‌ای (مانند «hawkish»، «bearish»، «guidance») شده و دقت طبقه‌بندی احساسات را افزایش می‌دهد.  
    \item \textbf{تنظیم مبتنی بر دستور} تعامل زبان طبیعی با اسناد مالی را تسهیل کرده و به عامل‌های یادگیری تقویتی یا تحلیل‌گران اجازه می‌دهد مدل‌ها را مستقیماً برای استخراج سیگنال‌های احساسی، نشانه‌های ریسک یا خلاصه‌های بازار به‌کار گیرند.  
\end{itemize}


\subsection{تحلیل احساسات در زمینه مالی\protect\footnotemark}
\footnotetext{\lr{Sentiment Analysis in Finance}}

\textbf{تحلیل احساسات}\footnote{\lr{Sentiment Analysis}} فرآیند استخراج اطلاعات ذهنی ــ مانند نظرات، نگرش‌ها یا هیجانات ــ از داده‌های متنی است. در حوزه مالی، تحلیل احساسات به‌دنبال کمی‌سازی لحن عاطفی مقالات خبری، گزارش‌های تحلیلی، تماس‌های اعلام درآمد یا بحث‌های شبکه‌های اجتماعی است؛ زیرا این منابع اغلب بر رفتار سرمایه‌گذاران و پویایی بازار اثرگذارند.  

\bigskip
\textbf{چرا احساسات اهمیت دارند:}  
بازارهای مالی صرفاً بر مبنای عوامل بنیادین هدایت نمی‌شوند، بلکه روان‌شناسی سرمایه‌گذاران و احساسات جمعی نیز نقشی تعیین‌کننده دارند~\cite{tetlock2007giving}. اخبار یا دیدگاه‌های مثبت می‌توانند حرکت‌های صعودی (\lr{Bullish}) را تقویت کنند، در حالی که احساسات منفی ممکن است به فروش گسترده منجر شود. با تبدیل متن‌های غیرساختاریافته به امتیازهای ساختاریافته احساسات، عامل‌های معاملاتی قادر خواهند بود سیگنال‌های روان‌شناختی را در فرآیند تصمیم‌گیری سیستماتیک وارد کنند.  

\bigskip
\textbf{روش‌های تحلیل احساسات:}
\begin{itemize}
    \item \textbf{رویکردهای واژگانی}\footnote{\lr{Lexicon-based Methods}}: مبتنی بر فرهنگ‌های لغت از پیش تعریف‌شده برای واژه‌های مثبت و منفی هستند. واژه‌نامه \lr{Loughran–McDonald}~\cite{loughran2011liability} در کاربردهای مالی بسیار رایج است زیرا واژگان ویژه این حوزه را پوشش می‌دهد.
    \item \textbf{یادگیری ماشین کلاسیک}\footnote{\lr{Classical Machine Learning}}: با استفاده از ویژگی‌های مهندسی‌شده مانند \textbf{کیسه واژگان}\footnote{\lr{Bag-of-Words}} یا بازنمایی‌های \lr{TF–IDF} و ترکیب آن‌ها با دسته‌بندهایی مانند \textbf{رگرسیون لجستیک}\footnote{\lr{Logistic Regression}} یا \textbf{ماشین‌های بردار پشتیبان}\footnote{\lr{Support Vector Machines, SVMs}}.
    \item \textbf{یادگیری عمیق و جاسازی‌های متنی}\footnote{\lr{Deep Learning \& Contextual Embeddings}}: مدل‌های مبتنی بر ترنسفورمر مانند \lr{BERT} و \lr{FinBERT}~\cite{araci2019finbert} بازنمایی‌های بافت‌محور از احساسات تولید می‌کنند که نسبت به روش‌های واژگانی و یادگیری سطحی عملکرد بسیار بهتری دارند.
    \item \textbf{مدل‌های زبانی بزرگ}\footnote{\lr{Large Language Models, LLMs}}: مدل‌های تنظیم‌شده بر اساس دستور (مانند \lr{InstructGPT} و \lr{ChatGPT}) می‌توانند طبقه‌بندی احساسات را در حالت \textbf{بدون نمونه}\footnote{\lr{Zero-shot}} یا \textbf{کم‌نمونه}\footnote{\lr{Few-shot}} انجام دهند و بدون نیاز به ریزتنظیم گسترده به بافت‌های مالی تطبیق یابند.
\end{itemize}

\bigskip
\textbf{ادغام سیگنال‌های احساسی در عامل‌های RL:}  
سیگنال‌های احساسی می‌توانند بخشی از بازنمایی حالت عامل‌های معاملاتی مبتنی بر \textbf{یادگیری تقویتی}\footnote{\lr{Reinforcement Learning, RL}} باشند:  
\begin{itemize}
    \item احساسات مثبت احتمال اتخاذ موقعیت‌های خرید (\lr{Long}) را افزایش می‌دهد.  
    \item احساسات منفی عامل را به سمت موقعیت‌های فروش (\lr{Short}) یا کاهش ریسک هدایت می‌کند.  
    \item احساسات خنثی یا نامطمئن می‌تواند راهبردهای محافظه‌کارانه یا پوششی (\lr{Hedging}) را تقویت کند.  
\end{itemize}

\bigskip
\textbf{چالش‌ها:}
\begin{itemize}
    \item زبان مالی دارای واژگان حوزه‌ای خاص است (برای نمونه: «beat expectations»، «hawkish»، «guidance») که ممکن است مدل‌های عمومی را گمراه کند.  
    \item طعنه، ابهام و ظرافت‌های زبانی هنوز به‌سختی توسط مدل‌ها شناسایی می‌شوند.  
    \item منابع خبری می‌توانند پرنویز، جانبدارانه یا دارای تأخیر باشند که قابلیت اطمینان سیگنال‌ها را کاهش می‌دهد.  
\end{itemize}

با وجود این چالش‌ها، تحلیل احساسات همچنان ابزاری قدرتمند برای تقویت راهبردهای معامله‌گری الگوریتمی باقی می‌ماند؛ به‌ویژه زمانی که در چارچوب‌های یادگیری تقویتی همراه با شاخص‌های فنی و داده‌های بازار به‌کار گرفته شود.  


\section{داده‌های بازار و شاخص‌های مالی}

\subsection{مبانی داده‌های OHLCV\protect\footnotemark}
\footnotetext{\lr{Open, High, Low, Close, Volume}}

\textbf{داده‌های OHLCV} نمایش استاندارد بازار مالی هستند که شامل قیمت باز، بیشینه، کمینه، بسته و حجم معاملات می‌شوند. این قالب، اساس تحلیل تکنیکال را تشکیل داده و به‌طور گسترده در معامله‌گری اختیاری و الگوریتمی به‌کار گرفته می‌شود.  

\begin{itemize}
    \item \textbf{Open (O):} نخستین قیمت معامله‌شده یک دارایی در ابتدای بازه زمانی مشخص (مانند یک دقیقه، یک ساعت یا یک روز).  
    \item \textbf{High (H):} بیشترین قیمتی که در طول بازه زمانی ثبت شده است.  
    \item \textbf{Low (L):} کمترین قیمتی که در طول بازه زمانی ثبت شده است.  
    \item \textbf{Close (C):} آخرین قیمت معامله‌شده دارایی در پایان بازه زمانی.  
    \item \textbf{Volume (V):} مجموع حجم معامله‌شده دارایی در طول بازه زمانی، که بیانگر میزان فعالیت و نقدشوندگی بازار است.  
\end{itemize}

\bigskip
\textbf{نمایش داده‌ها:}  
داده‌های OHLCV معمولاً با استفاده از \textbf{نمودار شمعی}\footnote{\lr{Candlestick Chart}} نمایش داده می‌شوند. در این نمودار، هر شمع قیمت‌های باز، بسته، بیشینه و کمینه را در یک بازه زمانی خلاصه می‌کند. ضخامت و رنگ بدنه شمع نشان‌دهنده صعودی بودن (\lr{Bullish}) یا نزولی بودن (\lr{Bearish}) قیمت در آن بازه است.  

\bigskip
\textbf{کاربردها:}
\begin{itemize}
    \item داده‌های OHLCV ورودی خام برای شاخص‌های فنی متداول مانند \textbf{میانگین‌های متحرک}\footnote{\lr{Moving Averages}}، \textbf{شاخص قدرت نسبی}\footnote{\lr{Relative Strength Index, RSI}} و \textbf{واگرایی/همگرایی میانگین متحرک}\footnote{\lr{Moving Average Convergence Divergence, MACD}} محسوب می‌شوند.  
    \item این داده‌ها هم \textbf{پویایی‌های قیمت} (از طریق O, H, L, C) و هم \textbf{میزان مشارکت بازار} (از طریق V) را منتقل می‌کنند.  
    \item بسته به افق معاملاتی و راهبرد، می‌توان از دقت‌های زمانی مختلف (داده تیک، درون‌روزی، روزانه یا هفتگی) استفاده کرد.  
    \item در چارچوب‌های یادگیری تقویتی، مقادیر OHLCV اغلب بخشی از فضای حالت هستند تا عامل بتواند از زمینه تاریخی ساختاریافته برای تصمیم‌گیری بهره ببرد.  
\end{itemize}

\bigskip
\textbf{محدودیت‌ها:}  
اگرچه OHLCV پویایی‌های اساسی قیمت و حجم را ثبت می‌کند، اما جزئیاتی مانند \textbf{عمق دفتر سفارش}\footnote{\lr{Order Book Depth}}، \textbf{شکاف خرید ـ فروش}\footnote{\lr{Bid–Ask Spread}} و نقدشوندگی پنهان را در بر نمی‌گیرد. از این رو، داده‌های OHLCV اغلب با منابع تکمیلی ــ مانند شاخص‌های احساسی یا داده‌های بنیادین ــ ترکیب می‌شوند تا قدرت پیش‌بینی و پایداری راهبردهای معامله‌گری الگوریتمی افزایش یابد.  

\subsection{شاخص‌های فنی پرکاربرد}

\textbf{شاخص‌های فنی}\footnote{\lr{Technical Indicators}} مقیاس‌های کمی مشتق‌شده از داده‌های \lr{OHLCV} هستند که معامله‌گران برای شناسایی \textbf{روندها}\footnote{\lr{Trends}}، \textbf{مومنتوم}\footnote{\lr{Momentum}}، \textbf{نوسان‌پذیری}\footnote{\lr{Volatility}} و نقاط احتمالی بازگشت به‌کار می‌گیرند. این شاخص‌ها هم در معامله‌گری اختیاری و هم در معامله‌گری الگوریتمی پرکاربردند و اغلب بخشی جدایی‌ناپذیر از فضای حالت برای عامل‌های یادگیری تقویتی محسوب می‌شوند.  

\bigskip
\textbf{شاخص‌های پیرو روند}\footnote{\lr{Trend-following Indicators}}:
\begin{itemize}
    \item \textbf{میانگین‌های متحرک}\footnote{\lr{Moving Averages, MA}}: میانگین قیمت‌های بسته‌شدن در یک پنجره زمانی ثابت. \textbf{میانگین متحرک ساده}\footnote{\lr{Simple Moving Average, SMA}} میانگینی بدون وزن است، در حالی که \textbf{میانگین متحرک نمایی}\footnote{\lr{Exponential Moving Average, EMA}} وزن بیشتری به قیمت‌های اخیر می‌دهد. تلاقی میانگین‌های کوتاه‌مدت و بلندمدت اغلب برای شناسایی تغییر روند استفاده می‌شود.  
    \item \textbf{واگرایی ـ همگرایی میانگین متحرک}\footnote{\lr{Moving Average Convergence Divergence, MACD}}: اختلاف میان دو EMA (معمولاً ۱۲ و ۲۶ دوره‌ای) تعریف می‌شود و یک خط سیگنال (EMA نه‌دوره‌ای) برای برجسته‌سازی تغییرات مومنتوم به آن افزوده می‌شود.  
\end{itemize}

\bigskip
\textbf{شاخص‌های تکانه}\footnote{\lr{Momentum Indicators}}:
\begin{itemize}
    \item \textbf{شاخص قدرت نسبی}\footnote{\lr{Relative Strength Index, RSI}}: یک نوسان‌نما در بازه ۰ تا ۱۰۰ است که بزرگی نسبی سودها در برابر زیان‌های اخیر را اندازه‌گیری می‌کند. مقادیر بالاتر از ۷۰ معمولاً شرایط اشباع خرید و مقادیر پایین‌تر از ۳۰ شرایط اشباع فروش را نشان می‌دهند.  
    \item \textbf{نوسان‌نمای تصادفی}\footnote{\lr{Stochastic Oscillator}}: قیمت بسته‌شدن را با دامنه قیمت‌ها در یک دوره بازگشت مشخص مقایسه می‌کند و بازتاب‌دهنده مومنتوم نسبت به بیشینه‌ها و کمینه‌های اخیر است.  
\end{itemize}

\bigskip
\textbf{شاخص‌های نوسان‌پذیری}\footnote{\lr{Volatility Indicators}}:
\begin{itemize}
    \item \textbf{باندهای بولینگر}\footnote{\lr{Bollinger Bands}}: با قرار دادن باندهای بالا و پایین در چند انحراف معیار از یک میانگین متحرک ساخته می‌شوند. این باندها در دوره‌های پرنوسان گسترش یافته و در دوره‌های کم‌نوسان منقبض می‌شوند.  
    \item \textbf{میانگین دامنه واقعی}\footnote{\lr{Average True Range, ATR}}: نوسان‌پذیری را با میانگین‌گیری از دامنه واقعی (بیشینه اختلاف‌های High–Low، High–Close و Low–Close) در یک دوره اندازه‌گیری می‌کند.  
\end{itemize}

\bigskip
\textbf{شاخص‌های مبتنی بر حجم}\footnote{\lr{Volume-based Indicators}}:
\begin{itemize}
    \item \textbf{حجم تعادلی}\footnote{\lr{On-Balance Volume, OBV}}: یک معیار تجمعی است که در روزهای صعودی حجم معاملات را اضافه و در روزهای نزولی کم می‌کند تا فشار خرید و فروش را بازتاب دهد.  
    \item \textbf{میانگین موزون حجمی قیمت}\footnote{\lr{Volume-Weighted Average Price, VWAP}}: میانگین قیمت معامله‌شده یک دارایی است که بر اساس حجم وزن‌دهی می‌شود و اغلب به‌عنوان معیار کیفیت اجرا توسط معامله‌گران نهادی استفاده می‌گردد.  
\end{itemize}

\bigskip
برای عامل‌های یادگیری تقویتی، شاخص‌های فنی بازنمایی‌های سطح بالاتری از داده‌های OHLCV فراهم می‌کنند و به عامل کمک می‌کنند الگوهایی مانند روندها، بازگشت‌ها یا رژیم‌های نوسان‌پذیری را که ممکن است در دنباله‌های خام قیمتی پنهان باشند شناسایی کند. ادغام این شاخص‌ها در فضای حالت می‌تواند کارایی یادگیری را از طریق تزریق دانش حوزه‌ای به مشاهدات عامل بهبود دهد.  

\bigskip
\textbf{محدودیت‌ها:}  
اگرچه شاخص‌های فنی می‌توانند بازنمایی حالت را غنی‌تر کنند، اما اتکای بیش از حد به آن‌ها می‌تواند منجر به فضاهای ویژگی با ابعاد بالا، نویز بیشتر و خطر \textbf{بیش‌برازش}\footnote{\lr{Overfitting}} شود. در عمل، انتخاب ویژگی محتاطانه، کاهش ابعاد یا استفاده از روش‌های یادگیری بازنمایی برای تضمین پایداری و مقاومت راهبردهای معاملاتی ضروری است.  


\subsection{چالش‌های موجود در بازارهای مالی}

به‌کارگیری \textbf{یادگیری تقویتی}\footnote{\lr{Reinforcement Learning, RL}} در بازارهای مالی ذاتاً دشوار است؛ زیرا این محیط پیچیده، پویا و پر از نویز است. برخلاف وظایف معیار کنترل‌شده، بازارها موانع متعددی ایجاد می‌کنند که طراحی عامل‌های معاملاتی مقاوم را با چالش روبه‌رو می‌سازد.  

\bigskip
\textbf{ناایستایی}\footnote{\lr{Non-stationarity}}:  
ویژگی‌های آماری سری‌های زمانی مالی به‌مرور تغییر می‌کنند؛ به دلیل تغییر رژیم‌های بازار، چرخه‌های اقتصادی، تحولات مقرراتی و شوک‌های پیش‌بینی‌نشده. راهبردی که در یک رژیم (مانند بازار صعودی) موفق عمل می‌کند، ممکن است در شرایط متفاوت (مانند رکود یا بحران‌ها) ناکام بماند.  

\bigskip
\textbf{نسبت بالای نویز به سیگنال}\footnote{\lr{High Noise-to-Signal Ratio}}:  
قیمت دارایی‌ها تحت تأثیر عوامل متنوع و غیرقابل‌پیش‌بینی ــ از جمله اعلامیه‌های کلان اقتصادی، رویدادهای ژئوپولیتیکی و روان‌شناسی سرمایه‌گذاران ــ قرار دارند. تشخیص الگوهای معنادار از نوسانات تصادفی کاری بسیار دشوار است.  

\bigskip
\textbf{مشاهده ناقص}\footnote{\lr{Partial Observability}}:  
معامله‌گران به‌ندرت به وضعیت کامل بازار دسترسی دارند. متغیرهای پنهان مانند جریان سفارش‌های نهادی، اطلاعات خصوصی و محرک‌های نهفته بازار تأثیر چشمگیری بر پویایی‌ها دارند، اما برای عامل قابل مشاهده نیستند.  

\bigskip
\textbf{پاداش‌های پراکنده و با تأخیر}\footnote{\lr{Sparse and Delayed Rewards}}:  
فرصت‌های سودآور نسبت به تعداد کنش‌های معاملاتی ممکن، به‌ندرت رخ می‌دهند. بنابراین سیگنال‌های پاداش هم پراکنده و هم با تأخیر هستند و مسئله تخصیص اعتبار در آموزش RL را پیچیده می‌سازند.  

\bigskip
\textbf{هزینه‌های تراکنش و لغزش قیمتی}\footnote{\lr{Transaction Costs and Slippage}}:  
اجرای واقعی معاملات مستلزم هزینه‌هایی مانند کمیسیون، \textbf{شکاف خرید ـ فروش}\footnote{\lr{Bid–Ask Spread}} و تأثیر بازار است. راهبردهایی که در شبیه‌سازی سودآور به نظر می‌رسند، ممکن است پس از در نظر گرفتن این اصطکاک‌ها دیگر کارایی نداشته باشند.  

\bigskip
\textbf{ریسک و رویدادهای دنباله‌ای}\footnote{\lr{Risk and Tail Events}}:  
بازده‌های مالی دارای دنباله‌های سنگین، خوشه‌بندی نوسان‌پذیری و رویدادهای شدید (مانند سقوط‌ها) هستند که بیش از آنچه مدل‌های گاوسی پیش‌بینی می‌کنند رخ می‌دهند. عامل‌های RL باید اهداف حساس به ریسک را لحاظ کنند تا از افت سرمایه‌های فاجعه‌بار جلوگیری شود.  

\bigskip
\textbf{محدودیت‌های داده}\footnote{\lr{Data Limitations}}:  
مجموعه‌داده‌های فرکانس بالا اغلب گران یا انحصاری هستند و دسترسی به آن‌ها محدود است. داده‌های عمومی (مانند داده‌های روزانه OHLCV) ممکن است برای راهبردهای کوتاه‌مدت یا درون‌روزی دقت کافی نداشته باشند.  

\bigskip
\textbf{سختی‌های ارزیابی}\footnote{\lr{Evaluation Difficulties}}:  
\textbf{بک‌تست}\footnote{\lr{Backtesting}} می‌تواند دچار \textbf{سوگیری نگاه به آینده}\footnote{\lr{Look-ahead Bias}}، \textbf{سوگیری بقا}\footnote{\lr{Survivorship Bias}} و \textbf{بیش‌برازش}\footnote{\lr{Overfitting}} نسبت به داده‌های تاریخی شود. اگرچه آزمون خارج از نمونه\footnote{\lr{Out-of-sample Testing}} ضروری است، اما ماهیت ناایستای بازارها تعمیم واقعی را به‌ویژه دشوار می‌سازد.  

\bigskip
این چالش‌ها نشان می‌دهند که طراحی دقیق سامانه‌های RL در مالی ضروری است. عامل‌های معاملاتی کارآمد باید سودآوری را با مدیریت ریسک متعادل سازند، هزینه‌های تراکنش را در نظر بگیرند، با رژیم‌های متغیر بازار تطبیق پیدا کنند و تحت چارچوب‌های ارزیابی سخت‌گیرانه و بدون سوگیری آزموده شوند. تنها در این صورت است که راهبردهای مبتنی بر RL می‌توانند به مقاومت و پایداری بلندمدت در بازارهای واقعی دست یابند.  


\subsection{شاخص‌ها و دارایی‌های مالی اصلی\protect\footnotemark}
\footnotetext{\lr{Major Financial Indices and Assets}}

عامل‌های \textbf{یادگیری تقویتی}\footnote{\lr{Reinforcement Learning, RL}} اغلب بر اساس شاخص‌های شناخته‌شده بازار و طبقات دارایی پرمعامله ارزیابی می‌شوند. این ابزارها نقدشوندگی بالایی دارند، به‌صورت جهانی شناخته‌شده‌اند و شرایط متنوعی از بازار را برای آزمون راهبردهای معامله‌گری الگوریتمی فراهم می‌کنند.  

\subsubsection{Dow Jones Industrial Average (DJIA)}
شاخص \lr{Dow Jones Industrial Average (DJIA)} یکی از قدیمی‌ترین و پرپیگیری‌ترین شاخص‌های سهام است که شامل ۳۰ شرکت بزرگ و عمومی ایالات متحده در صنایع مختلف می‌شود. اگرچه تنها بخش نسبتاً کوچکی از شرکت‌ها را پوشش می‌دهد، اما اغلب به‌عنوان معیاری برای عملکرد اقتصادی ایالات متحده در نظر گرفته می‌شود.  
برای عامل‌های معاملاتی، DJIA دسترسی به سهام بلوچیپ با \textbf{نوسان‌پذیری}\footnote{\lr{Volatility}} نسبتاً کمتر در مقایسه با شاخص‌های کوچک‌تر فراهم می‌آورد.  

\subsubsection{S\&P 500 Index}
شاخص \lr{S\&P 500} شامل ۵۰۰ شرکت بزرگ ایالات متحده است و به‌طور گسترده به‌عنوان بهترین معیار کلی بازار سهام آمریکا شناخته می‌شود. این شاخص حدود ۸۰\% از ارزش بازار ایالات متحده را پوشش می‌دهد.  
به دلیل نقدشوندگی بالا و پوشش گسترده، S\&P 500 یک معیار متداول برای ارزیابی عملکرد راهبردهای معامله‌گری الگوریتمی، از جمله عامل‌های RL، محسوب می‌شود.  

\subsubsection{XAU/USD}
\lr{XAU/USD} نشان‌دهنده قیمت طلا (XAU) بر حسب دلار آمریکا است. طلا هم یک کالا و هم یک دارایی پناهگاه امن محسوب می‌شود که در زمان عدم قطعیت اقتصادی یا تورم ارزش می‌یابد.  
برای عامل‌های معاملاتی، طلا چالشی منحصربه‌فرد ایجاد می‌کند زیرا دینامیک قیمتی آن تحت تأثیر شرایط کلان اقتصادی جهانی، سیاست‌های پولی و احساسات ریسکی قرار دارد.  

\subsubsection{رمزارزها\protect\footnotemark}
\footnotetext{\lr{Cryptocurrencies}}
رمزارزهایی مانند \lr{Bitcoin (BTC)} و \lr{Ethereum (ETH)} دارایی‌هایی بسیار پرنوسان و سفته‌بازانه هستند. برخلاف بازارهای سنتی، بازار رمزارزها به‌صورت ۲۴/۷ و در صرافی‌های غیرمتمرکز معامله می‌شوند و مجموعه‌داده‌های غنی‌ای برای آموزش RL فراهم می‌کنند.  
نوسان‌پذیری بالای آن‌ها هم فرصت و هم ریسک ایجاد می‌کند و آن‌ها را به بستر مناسبی برای آزمون عامل‌های RL با هدف شناسایی الگوهای کوتاه‌مدت قیمتی تبدیل می‌سازد.  

\subsubsection{بازار تبادل ارز خارجی\protect\footnotemark}
\footnotetext{\lr{Foreign Exchange Market, Forex}}
بازار \textbf{تبادل ارز خارجی} یا \textbf{فارکس} بزرگ‌ترین و نقدشونده‌ترین بازار مالی در جهان است، با حجم معاملات روزانه بیش از ۶ تریلیون دلار. جفت‌ارزهای اصلی (مانند \lr{EUR/USD}، \lr{USD/JPY} و \lr{GBP/USD}) بیشترین حجم معاملات را به خود اختصاص می‌دهند.  
بازار فارکس تحت تأثیر نرخ بهره، رویدادهای ژئوپولیتیکی و شاخص‌های کلان اقتصادی قرار دارد و بسیار پویاست. عامل‌های RL در فارکس باید با نوسانات سریع سازگار شوند و در عین حال هزینه‌های تراکنش و ریسک اهرم را نیز در نظر بگیرند.  

\bigskip
این شاخص‌ها و دارایی‌ها محیط‌های متنوعی را نمایندگی می‌کنند — از سهام (DJIA و S\&P 500) گرفته تا کالاها (XAU/USD)، دارایی‌های دیجیتال (رمزارزها) و ارزهای جهانی (فارکس). آزمون الگوریتم‌های RL در چنین ابزارهای گوناگونی موجب تعمیم وسیع‌تر، مقاومت در برابر تغییر رژیم‌ها و ارزیابی واقع‌بینانه‌تر در برابر معیارهای استاندارد مالی می‌شود.  


\section{مدیریت سبد دارایی‌ها و محدودیت‌های معامله‌گری}

\subsection{معامله تک‌سهم در برابر معامله سبد دارایی‌ها}

رویکردهای \textbf{یادگیری تقویتی}\footnote{\lr{Reinforcement Learning, RL}} در معامله‌گری الگوریتمی می‌توانند یا برای \textbf{یک دارایی منفرد} یا برای \textbf{یک سبد دارایی}\footnote{\lr{Portfolio}} به‌کار گرفته شوند. این انتخاب پیامدهای مهمی برای بازنمایی حالت، \textbf{فضای عمل}\footnote{\lr{Action Space}} و \textbf{طراحی پاداش}\footnote{\lr{Reward Design}} دارد.  

\bigskip
\textbf{معامله تک‌سهم}\footnote{\lr{Single-stock Trading}}:  
در این حالت، عامل RL تنها بر یک دارایی (مانند سهام \lr{Apple}، بیت‌کوین یا طلا) تمرکز می‌کند. فضای عمل می‌تواند گسسته (خرید، فروش، نگه‌داری) یا پیوسته (مانند تعیین اندازه موقعیت) باشد.  
\begin{itemize}
    \item \textbf{مزایا:} فضای حالت و عمل ساده‌تر، آموزش سریع‌تر و نیاز کمتر به داده.  
    \item \textbf{محدودیت‌ها:} نبود تنوع‌بخشی و تمرکز ریسک بر یک دارایی منفرد.  
\end{itemize}

\bigskip
\textbf{معامله سبد دارایی‌ها}\footnote{\lr{Portfolio Trading}}:  
در این حالت، عامل RL سرمایه را هم‌زمان میان چندین دارایی (مانند سهام، کالاها یا رمزارزها) تخصیص می‌دهد. فضای عمل چندبعدی است، زیرا عامل باید وزن‌های تخصیص برای هر دارایی را تعیین کند، مشروط به محدودیت‌های بودجه و ریسک.  
\begin{itemize}
    \item \textbf{مزایا:} امکان تنوع‌بخشی، واقع‌گرایی بیشتر برای سرمایه‌گذاری نهادی، و بهینه‌سازی بازدهی تعدیل‌شده بر اساس ریسک.  
    \item \textbf{محدودیت‌ها:} فضای حالت و عمل بزرگ‌تر، پیچیدگی آموزشی بالاتر و نیاز بیشتر به داده.  
\end{itemize}

\bigskip
به‌طور کلی، معامله تک‌سهم برای مطالعات \textbf{اثبات مفهوم}\footnote{\lr{Proof-of-Concept}} و مقایسه الگوریتم‌های RL ارزشمند است، در حالی که رویکردهای سبد دارایی بازتاب واقعی‌تری از مدیریت سرمایه دارند؛ جایی که تنوع‌بخشی و کنترل ریسک حیاتی است. به همین دلیل، پژوهش‌های نوین RL در مالی به‌طور فزاینده‌ای بر \textbf{بهینه‌سازی سبد دارایی‌ها}\footnote{\lr{Portfolio Optimization}} تمرکز می‌کنند تا راهبردهای معاملاتی مقاوم‌تر و واقع‌گرایانه‌تری ایجاد شود.  

\subsection{فضای حالت\protect\footnotemark}\label{subsec:state-space}
\footnotetext{\lr{State Space}}

در \textbf{یادگیری تقویتی}\footnote{\lr{Reinforcement Learning, RL}}، فضای حالت اطلاعاتی را تعریف می‌کند که در هر گام تصمیم‌گیری در دسترس عامل قرار دارد. در معامله‌گری، حالت بازنمایی محیط بازار، وضعیت \textbf{سبد دارایی‌ها}\footnote{\lr{Portfolio}} و گاه سیگنال‌های بیرونی را رمزگذاری می‌کند. طراحی فضای حالت مناسب یکی از حیاتی‌ترین جنبه‌های ساخت عامل معاملاتی RL است، زیرا تعیین می‌کند که عامل چه اطلاعاتی را می‌تواند مورد استفاده قرار دهد.  

\bigskip
\textbf{۱. تاریخچه قیمت و بازده:}  
ساده‌ترین بازنمایی حالت شامل مقادیر خام \lr{OHLCV} یا بازده‌های لگاریتمی در یک پنجره بازگشت است.  
\begin{itemize}
    \item \textbf{مزایا:} مستقیم، قابل تفسیر، و نیازمند کمترین پیش‌پردازش.  
    \item \textbf{محدودیت‌ها:} ناتوانی در ثبت وابستگی‌های زمانی پیچیده یا الگوهای مرتبه‌بالا.  
\end{itemize}

\bigskip
\textbf{۲. شاخص‌های فنی}\footnote{\lr{Technical Indicators}}:  
شاخص‌هایی مانند میانگین‌های متحرک، \lr{RSI}، \lr{MACD} و باندهای بولینگر اغلب به‌عنوان ویژگی‌های مهندسی‌شده از \lr{OHLCV} به حالت افزوده می‌شوند.  
\begin{itemize}
    \item \textbf{مزایا:} خلاصه‌سازی فشرده از پویایی قیمت، پرکاربرد در عمل.  
    \item \textbf{محدودیت‌ها:} استفاده بیش‌ازحد می‌تواند منجر به افزونگی و ریسک \textbf{بیش‌برازش}\footnote{\lr{Overfitting}} شود.  
\end{itemize}

\bigskip
\textbf{۳. اطلاعات سبد دارایی‌ها:}  
در وظایف \textbf{مدیریت سبد دارایی‌ها}\footnote{\lr{Portfolio Management}}، حالت معمولاً شامل تخصیص فعلی سرمایه میان دارایی‌ها، وجه نقد موجود و میزان استفاده از اهرم است.  
\begin{itemize}
    \item \textbf{مزایا:} تضمین می‌کند که تصمیم‌های معاملاتی بر اساس وضعیت واقعی سبد اتخاذ شوند.  
    \item \textbf{محدودیت‌ها:} در محیط‌های چنددارایی، ابعاد حالت افزایش می‌یابد.  
\end{itemize}

\bigskip
\textbf{۴. سیگنال‌های احساسی و بنیادی:}  
اخبار، گزارش‌های درآمد و رسانه‌های اجتماعی می‌توانند به امتیازهای احساسی تبدیل شوند و در حالت ادغام گردند.  
\begin{itemize}
    \item \textbf{مزایا:} ثبت محرک‌های حرکت قیمت فراتر از داده‌های فنی و کمک به توضیح تغییرات ناگهانی.  
    \item \textbf{محدودیت‌ها:} سیگنال‌های متنی پرنویز هستند و اغلب با تأخیر نسبت به حرکت قیمت همراه می‌شوند.  
\end{itemize}

\bigskip
\textbf{۵. داده‌های ریزساختار بازار}\footnote{\lr{Market Microstructure Data}}:  
در معامله‌گری با فرکانس بالا، حالت می‌تواند شامل عمق دفتر سفارش، \textbf{شکاف خرید ـ فروش}\footnote{\lr{Bid–Ask Spread}} و آمار عدم‌تعادل سفارش‌ها باشد.  
\begin{itemize}
    \item \textbf{مزایا:} ارائه بینش دقیق درباره نقدشوندگی درون‌روزی و پویایی کوتاه‌مدت بازار.  
    \item \textbf{محدودیت‌ها:} نیازمند داده‌های حجیم با فرکانس بالا و خطوط پردازش پیچیده.  
\end{itemize}

\bigskip
به‌طور کلی، فضای حالت بنیان اطلاعاتی عامل RL را تشکیل می‌دهد. بازنمایی‌های حداقلی ممکن است سیگنال‌های کلیدی را نادیده بگیرند، در حالی که بازنمایی‌های بیش‌ازحد پیچیده می‌توانند ریسک بیش‌برازش و ضعف تعمیم‌پذیری ایجاد کنند. در عمل، عامل‌های معاملاتی مقاوم اغلب از طراحی متوازن بهره می‌برند — ترکیبی از قیمت‌ها، شاخص‌های فنی، وضعیت سبد دارایی و سیگنال‌های احساسی — تا سازگاری در شرایط متنوع بازار تضمین شود.  

\subsection{شکل‌دهی پاداش\protect\footnotemark}\label{subsec:reward-shaping}
\footnotetext{\lr{Reward Shaping}}

در \textbf{یادگیری تقویتی}\footnote{\lr{Reinforcement Learning, RL}}، \textbf{تابع پاداش}\footnote{\lr{Reward Function}} 
رفتار عامل را هدایت می‌کند و تعیین می‌نماید که چه چیزی موفقیت تلقی می‌شود. در معامله‌گری، 
شکل‌دهی پاداش اهمیت ویژه‌ای دارد زیرا بازده‌های خام به‌تنهایی نمی‌توانند اهداف گسترده‌تری مانند 
کنترل ریسک، هزینه‌های تراکنش یا پایداری راهبرد را بازتاب دهند.  

\bigskip
\textbf{۱. پاداش‌های مبتنی بر سود}\footnote{\lr{Profit-based Rewards}}:  
مستقیم‌ترین رویکرد استفاده از بازده سبد دارایی به‌عنوان پاداش است:
\[
r_t = \frac{V_{t+1} - V_t}{V_t},
\]
که در آن $V_t$ ارزش سبد دارایی در زمان $t$ است.  
\begin{itemize}
    \item \textbf{مزایا:} هم‌راستایی مستقیم با بیشینه‌سازی سود.  
    \item \textbf{محدودیت‌ها:} نادیده‌گرفتن ریسک، که می‌تواند به راهبردهای پرنوسان منجر شود.  
\end{itemize}

\bigskip
\textbf{۲. پاداش‌های تعدیل‌شده بر اساس ریسک}\footnote{\lr{Risk-adjusted Rewards}}:  
پاداش‌ها می‌توانند با معیارهای ریسک (مانند نوسان‌پذیری، انحراف معیار نزولی یا افت سرمایه) نرمال شوند:
\[
r_t = \frac{R_t - R_f}{\sigma_t},
\]
که در آن $R_t$ بازده سبد دارایی، $R_f$ نرخ بدون ریسک و $\sigma_t$ برآورد نوسان‌پذیری است.  
\begin{itemize}
    \item \textbf{مزایا:} تشویق به راهبردهای روان‌تر و بازده‌های تعدیل‌شده بر اساس ریسک بالاتر.  
    \item \textbf{محدودیت‌ها:} نیازمند برآورد دقیق نوسان‌پذیری یا معیارهای ریسک.  
\end{itemize}

\bigskip
\textbf{۳. پاداش‌های مبتنی بر مطلوبیت}\footnote{\lr{Utility-based Rewards}}:  
گنجاندن توابع مطلوبیت (مانند لگاریتمی یا نمایی) امکان مدل‌سازی صریح ترجیحات ریسک را فراهم می‌کند:
\[
r_t = U(R_t) = \log(1 + R_t).
\]
\begin{itemize}
    \item \textbf{مزایا:} بازتاب‌دهنده ریسک‌گریزی سرمایه‌گذار و سازگار با نظریه اقتصادی.  
    \item \textbf{محدودیت‌ها:} انتخاب تابع مطلوبیت ذهنی و وابسته به بستر است.  
\end{itemize}

\bigskip
\textbf{۴. پاداش‌های چندهدفه}\footnote{\lr{Multi-objective Rewards}}:  
توابع پاداش می‌توانند بازده، هزینه‌های تراکنش و معیارهای ریسک را ترکیب کنند:
\[
r_t = R_t - \lambda_1 \cdot \text{Cost}_t - \lambda_2 \cdot \text{Risk}_t,
\]
که در آن $\lambda_1, \lambda_2$ تعادل میان سودآوری، اصطکاک‌های معاملاتی و ریسک را تعیین می‌کنند.  

\bigskip
شکل‌دهی پاداش مستقیماً رفتار معاملاتی را تعیین می‌کند:
\begin{itemize}
    \item پاداش‌های صرفاً مبتنی بر سود می‌توانند راهبردهای ناپایدار و پرریسک را تشویق کنند.  
    \item پاداش‌های تعدیل‌شده بر اساس ریسک یا مبتنی بر مطلوبیت، به پایداری و حفظ سرمایه کمک می‌کنند.  
    \item پاداش‌های چندهدفه امکان تعادل واقع‌گرایانه میان سود، ریسک و هزینه‌های اجرا را فراهم می‌سازند.  
\end{itemize}

بنابراین طراحی دقیق تابع پاداش برای عامل‌های RL که هدف آن‌ها استقرار در دنیای واقعی است، حیاتی محسوب می‌شود.  


\subsection{هزینه‌های تراکنش، لغزش قیمتی و محدودیت‌های نقدشوندگی\protect\footnotemark}\label{subsec:transaction-costs}
\footnotetext{\lr{Transaction Costs, Slippage, Liquidity Constraints}}

در عمل، راهبردهای معاملاتی با اصطکاک‌هایی مواجه‌اند که به‌شدت بر عملکرد اثر می‌گذارند. 
عامل‌های \textbf{یادگیری تقویتی}\footnote{\lr{Reinforcement Learning, RL}} که بدون درنظرگرفتن این عوامل آموزش می‌بینند، 
در معرض یادگیری سیاست‌های غیرواقعی قرار دارند که در بازارهای واقعی شکست می‌خورند.  

\bigskip
\textbf{هزینه‌های تراکنش}\footnote{\lr{Transaction Costs}}:  
هر معامله شامل هزینه‌های صریحی مانند کارمزد کارگزار، هزینه‌های بورس و شکاف خرید ـ فروش\footnote{\lr{Bid–Ask Spread}} است. 
در محیط‌های RL، این هزینه‌ها معمولاً به‌صورت کارمزدی متناسب با حجم معامله مدل می‌شوند:
\[
\text{Cost}_t = c \cdot |q_t|,
\]
که در آن $c$ نرخ هزینه و $q_t$ اندازه معامله است. 
نادیده‌گرفتن هزینه‌های تراکنش اغلب منجر به بیش‌معامله‌گری می‌شود.  

\bigskip
\textbf{لغزش قیمتی}\footnote{\lr{Slippage}}:  
لغزش به اختلاف میان قیمت مورد انتظار اجرا و قیمت واقعی پرشده اشاره دارد که معمولاً ناشی از 
بازارهای پرسرعت یا عمق محدود دفتر سفارش است. 
لغزش معمولاً با افزایش اندازه سفارش و نوسان‌پذیری بیشتر می‌شود.  
در شبیه‌سازی‌ها، لغزش اغلب به‌صورت یک هزینه اضافی متناسب با حجم معامله و نوسان‌پذیری مدل می‌شود.  

\bigskip
\textbf{محدودیت‌های نقدشوندگی}\footnote{\lr{Liquidity Constraints}}:  
نقدشوندگی تعیین می‌کند که یک دارایی تا چه حد می‌تواند بدون اثرگذاری چشمگیر بر قیمت معامله شود. 
دارایی‌های بسیار نقدشونده (مانند سهام شاخص \lr{S\&P 500} یا جفت‌های اصلی \lr{Forex}) 
می‌توانند سفارش‌های بزرگ را با لغزش حداقلی جذب کنند،
در حالی که دارایی‌های کم‌نقدشونده (مانند سهام شرکت‌های کوچک یا برخی رمزارزها) بیشتر در معرض اثرگذاری بازار هستند.  
در محیط‌های RL، نقدشوندگی می‌تواند به‌صورت‌های زیر مدل شود:
\begin{itemize}
    \item سقف‌گذاری بر اندازه موقعیت یا حجم معامله نسبت به حجم بازار،  
    \item افزودن تأخیر در اجرا برای معاملات بزرگ،  
    \item جریمه‌کردن گردش بیش‌ازحد از طریق تابع پاداش.  
\end{itemize}

\bigskip
\begin{itemize}
    \item بدون مدل‌سازی هزینه‌ها و لغزش، عامل‌های RL ممکن است به بسامدهای معاملاتی غیرواقعی (مانند صدها معامله در روز) تکیه کنند.  
    \item محدودیت‌های آگاه به نقدشوندگی تضمین می‌کنند که عامل‌ها راهبردهایی مقیاس‌پذیر و مناسب برای استفاده نهادی بیاموزند.  
    \item یکپارچه‌سازی این اصطکاک‌ها در تابع پاداش یا در پویایی محیط به نتایجی مقاوم‌تر و واقع‌گرایانه‌تر در معامله‌گری منجر می‌شود.  
\end{itemize}


\subsection{معیارهای ریسک\protect\footnotemark}\label{subsec:reward-shaping}
\footnotetext{\lr{Risk Measures}}

در معامله‌گری، سودآوری همواره باید در بستر ریسک ارزیابی شود. 
عامل‌های \textbf{یادگیری تقویتی}\footnote{\lr{Reinforcement Learning, RL}} که ریسک را نادیده می‌گیرند، 
ممکن است به سیاست‌های ناپایدار یا غیرعملی همگرا شوند. 
\textbf{معیارهای ریسک}\footnote{\lr{Risk Measures}} ابزارهای کمی برای ارزیابی و محدود کردن 
مواجهه نزولی یک راهبرد معاملاتی فراهم می‌کنند.  

\bigskip
\textbf{۱. نوسان‌پذیری}\footnote{\lr{Volatility, Standard Deviation}}:  
رایج‌ترین معیار ریسک است و پراکندگی بازده‌ها را ثبت می‌کند:
\[
\sigma = \sqrt{\frac{1}{T-1}\sum_{t=1}^T (R_t - \bar{R})^2},
\]
که در آن $R_t$ بازده در زمان $t$ است.  
\begin{itemize}
    \item \textbf{مزایا:} محاسبه ساده و استفاده گسترده در نسبت‌های عملکردی مانند نسبت شارپ.  
    \item \textbf{محدودیت‌ها:} انحراف‌های مثبت و منفی را یکسان جریمه می‌کند، 
    در حالی‌که سرمایه‌گذاران معمولاً بیشتر به ریسک نزولی اهمیت می‌دهند.  
\end{itemize}

\bigskip
\textbf{۲. ارزش در معرض ریسک}\footnote{\lr{Value at Risk, VaR}}:  
معیاری که حداکثر زیان بالقوه در یک افق زمانی مشخص با سطح اطمینان $\alpha$ را برآورد می‌کند:
\[
\text{VaR}_{\alpha}(T) = \inf \{ x \in \mathbb{R} : P(L_T > x) \leq 1-\alpha \},
\]
که در آن $L_T$ زیان سبد دارایی است.  
\begin{itemize}
    \item \textbf{مزایا:} استفاده گسترده در مدیریت ریسک نهادی و مقرراتی.  
    \item \textbf{محدودیت‌ها:} شدت زیان‌های فراتر از آستانه VaR را ثبت نمی‌کند.  
\end{itemize}

\bigskip
\textbf{۳. ارزش شرطی در معرض ریسک}\footnote{\lr{Conditional VaR, CVaR / Expected Shortfall}}:  
زیان مورد انتظار را مشروط به فراتررفتن از VaR اندازه‌گیری می‌کند:
\[
\text{CVaR}_{\alpha} = \mathbb{E}[L_T \mid L_T > \text{VaR}_{\alpha}].
\]
\begin{itemize}
    \item \textbf{مزایا:} معیار ریسک همدوس و پوشش‌دهنده ریسک‌های دنباله‌ای و زیان‌های شدید.  
    \item \textbf{محدودیت‌ها:} برآورد محاسباتی سنگین‌تر، به‌ویژه در محیط‌های با فرکانس بالا.  
\end{itemize}

\bigskip
\textbf{۴. بیشینه افت سرمایه}\footnote{\lr{Maximum Drawdown, MDD}}:  
بزرگ‌ترین افت اوج تا حضیض در ارزش سبد طی یک افق زمانی.  
\begin{itemize}
    \item \textbf{مزایا:} شهودی و پرکاربرد در عمل برای ارزیابی بدترین زیان‌ها.  
    \item \textbf{محدودیت‌ها:} فرکانس یا مدت افت‌ها را نادیده می‌گیرد و تفسیر احتمالاتی ندارد.  
\end{itemize}

\bigskip
\textbf{۵. معیارهای ریسک نزولی}\footnote{\lr{Downside Risk Measures}}:  
مانند انحراف نزولی (استفاده‌شده در نسبت \lr{Sortino}) تنها بر بازده‌های منفی نسبت به بازده هدف یا حداقل قابل‌قبول تمرکز دارند.  
\begin{itemize}
    \item \textbf{مزایا:} هم‌راستا با ریسک‌گریزی سرمایه‌گذار؛ جریمه‌کردن صرفاً نوسان‌های مضر.  
    \item \textbf{محدودیت‌ها:} نیازمند تعیین بازده هدف است که ممکن است ذهنی باشد.  
\end{itemize}

\bigskip
\begin{itemize}
    \item معیارهای ریسک می‌توانند در تابع پاداش ادغام شوند (مانند پاداش‌های مبتنی بر نسبت شارپ یا نسبت Sortino).  
    \item آن‌ها می‌توانند به‌عنوان قیود صریح در بهینه‌سازی سبد دارایی استفاده شوند (مثلاً محدودکردن CVaR یا MDD).  
    \item پایش معیارهای ریسک در طول آموزش تضمین می‌کند که عامل‌های RL راهبردهایی بیاموزند که نه‌تنها سودآور بلکه مقاوم و قابل‌استقرار در محیط واقعی باشند.  
\end{itemize}
\subsection{معیارهای ارزیابی\protect\footnotemark}\label{subsec:evaluation-metrics}
\footnotetext{\lr{Evaluation Metrics}}

ارزیابی عامل‌های \textbf{یادگیری تقویتی}\footnote{\lr{Reinforcement Learning, RL}} در معامله‌گری نیازمند معیارهایی است که سودآوری را همراه با ریسک و پایداری در نظر بگیرند. اتکا به بازده خام می‌تواند منجر به راهبردهای ناپایدار یا غیرعملی شود؛ ازاین‌رو معمولاً ترکیبی از معیارهای تعدیل‌شده بر اساس ریسک، سنجه‌های رشد و معیارهای تشخیصی به‌کار می‌روند.  

\subsubsection{Sharpe Ratio و Sortino Ratio}

سودآوری به‌تنهایی برای ارزیابی یک راهبرد در بازارهای پرنوسان کافی نیست. معیارهای تعدیل‌شده بر اساس ریسک مانند \textbf{Sharpe Ratio}\footnote{Sharpe Ratio} و \textbf{Sortino Ratio}\footnote{Sortino Ratio} به‌طور گسترده برای سنجش عملکرد نسبت به ریسک استفاده می‌شوند.  

\bigskip
\textbf{Sharpe Ratio:}  
این نسبت بازده مازاد کسب‌شده به ازای هر واحد ریسک کل (انحراف معیار بازده‌ها) را اندازه‌گیری می‌کند:
\[
\text{Sharpe Ratio} = \frac{R_p - R_f}{\sigma_p},
\]
که در آن $R_p$ میانگین بازده سبد دارایی، $R_f$ نرخ بدون ریسک و $\sigma_p$ انحراف معیار بازده‌ها است.  
\begin{itemize}
    \item \textbf{تفسیر:} مقادیر بالاتر Sharpe Ratio نشان‌دهنده عملکرد بهتر تعدیل‌شده بر اساس ریسک هستند، هرچند این معیار نوسان‌های مثبت و منفی را یکسان جریمه می‌کند.  
\end{itemize}

\bigskip
\textbf{Sortino Ratio:}  
این نسبت اصلاحی بر Sharpe Ratio است که تنها بر نوسان نزولی تمرکز دارد:
\[
\text{Sortino Ratio} = \frac{R_p - R_f}{\sigma_d},
\]
که در آن $\sigma_d$ انحراف نزولی نسبت به بازده حداقلی قابل‌قبول (\lr{Minimum Acceptable Return, MAR}) است.  
\begin{itemize}
    \item \textbf{تفسیر:} میان نوسان‌های مضر (زیان‌ها) و مفید (سودها) تمایز قائل می‌شود و اغلب در بستر معامله‌گری مناسب‌تر است.  
\end{itemize}

\bigskip
برای عامل‌های RL، Sharpe Ratio و Sortino Ratio معیارهای کلیدی برای متوازن‌سازی سود و ریسک محسوب می‌شوند و معمولاً هم به‌عنوان ابزار ارزیابی و هم به‌عنوان بخشی از طراحی پاداش برای تشویق سیاست‌های پایدار به‌کار می‌روند.  

\subsubsection{Compound Annual Growth Rate (CAGR)\protect\footnotemark}
\footnotetext{\lr{Compound Annual Growth Rate, CAGR}}

CAGR رشد میانگین سالانه یک سرمایه‌گذاری را با فرض بازسرمایه‌گذاری سودها اندازه‌گیری می‌کند:
\[
\text{CAGR} = \left( \frac{V_f}{V_i} \right)^{\tfrac{1}{T}} - 1,
\]
که در آن $V_f$ ارزش نهایی سبد، $V_i$ ارزش اولیه و $T$ تعداد سال‌ها است.  

\begin{itemize}
    \item \textbf{تفسیر:} CAGR شاخصی هموار از رشد بلندمدت ارائه می‌دهد و اثر نوسان‌پذیری بر میانگین بازده را حذف می‌کند.  
\end{itemize}

برای عامل‌های RL، CAGR پایداری یک راهبرد را در افق‌های بلندمدت برجسته می‌کند؛ حتی رشدهای متوسط ولی پایدار می‌توانند بر راهبردهای پرنوسان با بازده‌های کوتاه‌مدت بیشتر برتری داشته باشند.  

\subsubsection{Maximum Drawdown (MDD)\protect\footnotemark}
\footnotetext{\lr{Maximum Drawdown, MDD}}

MDD بزرگ‌ترین افت اوج تا حضیض ارزش سبد را پیش از رسیدن به یک اوج جدید اندازه‌گیری می‌کند:
\[
\text{MDD} = \max_{t \in [0,T]} \left( \frac{V_{\text{peak up to } t} - V_t}{V_{\text{peak up to } t}} \right).
\]

\begin{itemize}
    \item \textbf{تفسیر:} مقادیر کوچک‌تر MDD نشان‌دهنده راهبردهای باثبات‌تر هستند؛ افت‌های بزرگ مشکل‌سازند زیرا جبران آن‌ها نیازمند سودهای نامتناسب است.  
\end{itemize}

MDD برای ارزیابی عامل‌های RL حیاتی است، چراکه افت‌های شدید می‌توانند راهبردی سودآور را در عمل غیرقابل‌قبول سازند. این معیار اغلب همراه با Sharpe Ratio و CAGR برای یک ارزیابی متوازن استفاده می‌شود.  

\subsubsection{نرخ برد\protect\footnotemark \; و دقت معاملات\protect\footnotemark}
\footnotetext{\lr{Win Rate}}
\footnotetext{\lr{Trade Accuracy}}

\textbf{Win Rate} نسبت معاملات سودآور به کل معاملات است:
\[
\text{Win Rate} = \frac{\text{تعداد معاملات سودآور}}{\text{کل معاملات}} \times 100\%.
\]

\textbf{Trade Accuracy} میزان درستی عامل در پیش‌بینی جهت حرکت قیمت را اندازه‌گیری می‌کند:
\[
\text{Accuracy} = \frac{\text{تعداد پیش‌بینی‌های صحیح جهت قیمت}}{\text{کل پیش‌بینی‌ها}} \times 100\%.
\]

\begin{itemize}
    \item \textbf{تفسیر:} نرخ برد یا دقت بالا لزوماً تضمین‌کننده سودآوری نیست، زیرا زیان‌های بزرگ می‌توانند سود معاملات موفق را خنثی کنند. برعکس، حتی یک راهبرد با Win Rate پایین ممکن است سودآور باشد اگر میانگین سود معاملات برد از زیان‌ها بیشتر باشد.  
\end{itemize}

برای عامل‌های RL، Win Rate و Trade Accuracy بیشتر نقش تشخیصی دارند و کیفیت پیش‌بینی یا تصمیم‌گیری را نشان می‌دهند. بااین‌حال، بیشترین ارزش آن‌ها زمانی است که در کنار معیارهای تعدیل‌شده بر اساس ریسک (مانند Sharpe و Sortino)، سنجه‌های رشد (CAGR) و معیارهای ریسک نزولی (MDD) به کار روند تا یک چارچوب ارزیابی جامع حاصل شود.

\lr{FinGPT} از مدل‌های زبانی بزرگ متن‌باز استفاده کرده و آن‌ها را با بهره‌گیری از ریزتنظیم\footnote{Fine-tuning} کارآمد و تکنیک‌های هم‌ترازی به حوزه مالی سازگار می‌کند:
\begin{itemize}
    \item \textbf{ستون فقرات مدل\footnote{Model Backbone}:} ساخته‌شده بر پایه معماری‌های متن‌باز خانواده \lr{GPT}.  
    \item \textbf{انطباق حوزه‌ای:} استفاده از تنظیم مبتنی بر دستورالعمل\footnote{Instruction Tuning} و ریزتنظیم بر پیکره مالی برای سازگاری با وظایف خاص حوزه.  
    \item \textbf{کارایی:} به‌کارگیری روش‌های سازگار کم‌رتبه (\lr{LoRA}) و رویکردهای سبک و بهینه‌سازی پارامترها برای کاهش هزینه محاسباتی.  
    \item \textbf{هم‌ترازی:} بهره‌گیری از یادگیری تقویتی با بازخورد انسانی\footnote{Reinforcement Learning with Human Feedback, RLHF} برای اطمینان از پاسخ‌های قابل‌اعتماد و متناسب با وظایف مالی.  
\end{itemize}

    \item قابلیت سازگاری در طیف متنوعی از وظایف شامل طبقه‌بندی احساسات، پرسش و پاسخ و خلاصه‌سازی.

\subsection{Stable-Baselines3: Reliable Reinforcement Learning Implementations \cite{raffin2021stable}}

\subsubsection{تمرکز پژوهش}
\lr{Stable-Baselines3 (SB3)} یک کتابخانه متن‌باز است که پیاده‌سازی‌های قابل‌اعتماد و آزموده‌شده از الگوریتم‌های یادگیری تقویتی عمیق\footnote{Deep Reinforcement Learning, DRL} در پایتون ارائه می‌دهد. این کتابخانه که بر پایه \lr{PyTorch} و به‌عنوان جانشین \lr{Stable-Baselines} (مبتنی بر \lr{TensorFlow}) ساخته شده است، هدف اصلی‌اش تضمین بازتولیدپذیری، سهولت استفاده و قابلیت توسعه پژوهشی در یادگیری تقویتی است.  
این کتابخانه با ارائه پیاده‌سازی‌های استاندارد الگوریتم‌ها برای بنچمارک‌گیری\footnote{Benchmarking}، آزمایش و توسعه روش‌های جدید، هم از پژوهشگران و هم از متخصصان کاربردی پشتیبانی می‌کند.

\subsubsection{روش‌شناسی}
\lr{SB3} بر قابلیت اعتماد، بازتولیدپذیری و دسترس‌پذیری از طریق اصول طراحی زیر تأکید می‌کند:
\begin{itemize}
    \item \textbf{قابلیت اعتماد و بازتولیدپذیری:} اجرای پروتکل‌های تست سخت‌گیرانه (واحد، یکپارچه‌سازی، بازگشتی) و تضمین نتایج قطعی در صورت امکان.
    \item \textbf{سهولت استفاده:} ارائه یک رابط سطح بالا و کاربرپسند (مانند \lr{\texttt{.learn()}}) با حلقه‌های آموزشی و ارزیابی سازگار در تمام الگوریتم‌ها. سازگاری کامل با \lr{OpenAI Gym}، \lr{PyTorch} و \lr{NumPy}.
    \item \textbf{قابلیت توسعه:} معماری ماژولار که اجازه می‌دهد سیاست‌ها، شبکه‌ها، حافظه‌های بازپخش\footnote{Replay Buffers} و توابع زیان به‌راحتی سفارشی شوند، و این امکان را فراهم می‌کند که نمونه‌سازی سریع روش‌های جدید RL صورت گیرد.
\end{itemize}

\subsubsection{یافته‌ها}
\begin{itemize}
    \item \lr{SB3} شامل الگوریتم‌های پرکاربرد یادگیری تقویتی است، هم مبتنی بر سیاست جاری\footnote{On-policy} (مانند PPO, A2C, TRPO) و هم خارج از سیاست\footnote{Off-policy} (مانند DQN, DDPG, TD3, SAC).  
    \item بنچمارک‌گیری در محیط‌های استاندارد (مانند \lr{MuJoCo} و \lr{Atari}) نتایجی هم‌تراز یا بهتر از آنچه در مقالات اصلی الگوریتم‌ها و نسخه قبلی \lr{Stable-Baselines} گزارش شده بود ارائه داد.  
    \item ابرپارامترها\footnote{Hyperparameters} و پیکربندی‌های پیش‌فرض با دقت بر اساس مقالات اصلی تنظیم شده‌اند تا نتایج قابل‌اعتماد حاصل شود.  
\end{itemize}

\subsubsection{محدودیت‌ها}
\begin{itemize}
    \item تمرکز بر پیاده‌سازی‌های تک‌ماشینه و غیرتوزیعی؛ فاقد مقیاس‌پذیری مشابه چارچوب‌هایی مانند \lr{RLlib} یا \lr{Tianshou}.  
    \item الگوریتم‌های آزمایشی به‌صورت جداگانه در \lr{SB3-Contrib} ارائه می‌شوند که ممکن است همان سطح پایداری و آزمون‌پذیری را نداشته باشند.  
    \item با وجود اولویت‌دهی به بازتولیدپذیری، نتایج همچنان ممکن است بسته به تصادفی بودن محیط یا تفاوت سخت‌افزار تغییر کنند.  
\end{itemize}

\subsection{FinRL: A Deep Reinforcement Learning Library for Automated Stock Trading in Quantitative Finance \cite{liu2020finrl}}

\subsubsection{تمرکز پژوهش}
\lr{FinRL} یکی از نخستین کتابخانه‌های متن‌باز یادگیری تقویتی عمیق\footnote{Deep Reinforcement Learning, DRL} است که به‌طور خاص برای حوزه مالی کمی طراحی شده است. هدف اصلی آن فراهم‌کردن یک چارچوب بازتولیدپذیر، قابل‌گسترش و کاربرپسند است که سد ورود برای پژوهشگران و متخصصان علاقه‌مند به توسعه راهبردهای معامله‌گری خودکار را کاهش دهد.  

\subsubsection{روش‌شناسی}
این کتابخانه یک \textbf{معماری سه‌لایه ماژولار} اتخاذ می‌کند که انعطاف‌پذیری و قابلیت توسعه را در کاربردهای مختلف ممکن می‌سازد:
\begin{itemize}
    \item \textbf{لایه محیط:} شبیه‌سازی بازار با استفاده از داده‌های تاریخی یا بلادرنگ، شامل هزینه‌های معاملاتی، محدودیت‌های نقدشوندگی و ترجیحات ریسک سرمایه‌گذار.  
    \item \textbf{لایه عامل:} پیاده‌سازی الگوریتم‌های پیشرفته DRL (مانند \lr{DQN, DDPG, PPO, SAC, A2C, TD3}) با توابع پاداش قابل‌سفارشی‌سازی.  
    \item \textbf{لایه کاربرد:} ماژول‌های آماده برای معامله سهام، تخصیص دارایی و معامله رمزارز، همراه با ابزارهای یکپارچه برای بک‌تست\footnote{Backtesting}.  
\end{itemize}

این چارچوب در \lr{PyTorch} پیاده‌سازی شده، با \lr{OpenAI Gym}-API سازگار است و مستندات و آموزش‌های گسترده‌ای برای پشتیبانی از آموزش و بازتولیدپذیری ارائه می‌دهد.

    \item به‌طور گسترده در پژوهش‌های دانشگاهی و آموزش مورد استفاده قرار گرفته و پلی میان یادگیری ماشین و مالی کمی ایجاد کرده است.

\subsection{Deep Reinforcement Learning for Automated Stock Trading: An Ensemble Strategy \cite{yang2020ensemble}}

\subsubsection{تمرکز پژوهش}
این مطالعه یک راهبرد \textit{تجمیعی}\footnote{Ensemble Strategy} مبتنی بر یادگیری تقویتی عمیق\footnote{Deep Reinforcement Learning, DRL} را برای معامله‌گری خودکار سهام پیشنهاد می‌دهد. هدف اصلی، ترکیب نقاط قوت چند الگوریتم DRL برای ساخت یک عامل معاملاتی مقاوم‌تر و سازگارتر است که بتواند با رژیم‌های پویا و پرنوسان بازار مقابله کند.

\subsubsection{روش‌شناسی}
چارچوب تجمیعی چند الگوریتم \textit{بازیگر–منتقد}\footnote{Actor–Critic} را با یک فرآیند انتخاب سیستماتیک ادغام می‌کند:
\begin{itemize}
    \item \textbf{الگوریتم‌های استفاده‌شده:} \lr{Proximal Policy Optimization (PPO)}، \lr{Advantage Actor–Critic (A2C)} و \lr{Deep Deterministic Policy Gradient (DDPG)}.  
    \item \textbf{تکنیک بارگذاری در صورت نیاز\footnote{Load-on-demand technique}:} روشی کارآمد برای مدیریت داده با هدف کاهش سربار حافظه در فضای عمل پیوسته.  
    \item \textbf{سازوکار تجمیع:}  
    \begin{itemize}
        \item عامل‌ها با استفاده از پنجره‌های متحرک داده‌های تاریخی آموزش داده می‌شوند.  
        \item یک پنجره اعتبارسنجی برای ارزیابی عامل‌ها بر اساس نسبت شارپ\footnote{Sharpe Ratio} (بازده تعدیل‌شده بر اساس ریسک) و شاخص آشفتگی\footnote{Turbulence Index} (کنترل ریسک) استفاده می‌شود.  
        \item بهترین عامل برای اجرای معاملات در دوره معاملاتی بعدی انتخاب می‌شود.  
    \end{itemize}
    \item \textbf{مجموعه‌داده:} سی سهم از شاخص \lr{Dow Jones Industrial Average (DJIA)}.  
    \item \textbf{مقایسه با مبنا:} شاخص DJIA و پرتفوی حداقل واریانس\footnote{Minimum-variance Portfolio}.  
\end{itemize}

\subsection{A Novel Deep Reinforcement Learning-Based Automated Stock Trading System Using Cascaded LSTM Networks \cite{zou2022cascaded}}

\subsubsection{تمرکز پژوهش}
این مقاله به چالش‌های به‌کارگیری یادگیری تقویتی عمیق\footnote{Deep Reinforcement Learning, DRL} در سری‌های زمانی مالی می‌پردازد؛ سری‌هایی که با نسبت پایین سیگنال به نویز و الگوهای نامنظم و نایستای مشخص می‌شوند. نویسندگان یک معماری \lr{Cascaded LSTM} پیشنهاد می‌کنند تا توانایی عامل‌های DRL در ثبت وابستگی‌های زمانی و ساختارهای پنهان در داده‌های روزانه قیمت سهام افزایش یابد.

\subsubsection{روش‌شناسی}
این چارچوب با یک طراحی دو‌مرحله‌ای \lr{Cascaded LSTM}، استخراج ویژگی و یادگیری تقویتی را ترکیب می‌کند:
\begin{itemize}
    \item \textbf{مرحله استخراج ویژگی:} نخستین شبکه \lr{LSTM} الگوهای زمانی را از داده‌های خام روزانه سهام مدل‌سازی می‌کند.  
    \item \textbf{مرحله عامل:} ویژگی‌های استخراج‌شده به یک عامل DRL مبتنی بر \lr{Proximal Policy Optimization (PPO)} داده می‌شوند، جایی که یک LSTM دیگر درون شبکه‌های \textit{بازیگر–منتقد}\footnote{Actor–Critic} تعبیه شده تا تصمیم‌گیری دنباله‌ای بهبود یابد.  
\end{itemize}

\noindent این سیستم بر روی چهار شاخص عمده سهام ارزیابی شده است: \lr{Dow Jones Industrial Average (DJI, US)}، \lr{SSE50 (China)}، \lr{SENSEX (India)} و \lr{FTSE100 (UK)}. عملکرد آن در برابر مبناهایی مانند \textit{Buy-and-Hold}، \lr{PPO-MLP}، مدل‌های \lr{Standalone LSTM/MLP}، \lr{LightGBM}، \lr{Gradient Boosting} و راهبردهای DRL تجمیعی مقایسه شده است.

    \item عملکردی بهتر از مبناهای یادگیری ماشین سنتی و مدل‌های DRL بدون ادغام \lr{Cascaded LSTM} داشت.  
    \item در بازار \lr{SSE50} عملکرد به‌ویژه قوی بود که نشان‌دهنده سازگاری در زمینه بازارهای نوظهور است.

\subsection{Commission Fee is not Enough: A Hierarchical Reinforced Framework for Portfolio Management \cite{wang2021commission}}

\subsubsection{تمرکز پژوهش}
این مطالعه به یک محدودیت رایج در مدیریت پرتفوی مبتنی بر یادگیری تقویتی عمیق\footnote{Deep Reinforcement Learning, DRL} اشاره می‌کند: اغلب مدل‌های موجود تنها کارمزد کمیسیون را به‌عنوان هزینه معاملاتی در نظر می‌گیرند و تأثیر قابل‌توجه \textbf{لغزش قیمتی\footnote{Price Slippage}} را نادیده می‌گیرند. از آنجا که لغزش می‌تواند نتایج اجرا را به‌طور چشمگیری تغییر دهد، نویسندگان \textbf{HRPM} را پیشنهاد می‌کنند؛ یک چارچوب تقویتی سلسله‌مراتبی که تخصیص پرتفوی را از اجرای معاملات به‌طور صریح جدا می‌کند.

\subsubsection{روش‌شناسی}
چارچوب HRPM از طراحی سلسله‌مراتبی دو‌سطحی استفاده می‌کند:
\begin{itemize}
    \item \textbf{سیاست سطح بالا:} وزن‌های پرتفوی را در فرکانس پایین‌تر تعیین می‌کند با هدف بیشینه‌سازی بازده بلندمدت.  
    \item \textbf{سیاست سطح پایین:} معاملات را در بازه‌های کوتاه‌تر اجرا می‌کند و به‌طور راهبردی هزینه‌های معاملاتی از جمله لغزش را به حداقل می‌رساند.  
\end{itemize}

راهبرد آموزشی شامل موارد زیر است:
\begin{itemize}
    \item پیش‌آموزش هر دو سطح به‌صورت جداگانه برای ایجاد رفتارهای پایه.  
    \item آموزش مشترک تکرارشونده برای دستیابی به کارایی هماهنگ بین راهبرد و اجرا.  
\end{itemize}

این چارچوب بر روی بازار سهام ایالات متحده و چین ارزیابی شده و با خط‌مبناهای DRL و روش‌های سنتی مدیریت پرتفوی مقایسه شده است.

\subsubsection{یافته‌ها}
\begin{itemize}
    \item \textbf{هزینه‌های معاملاتی:} تحلیل نشان می‌دهد که کارمزد کمیسیون تنها بخش کوچکی از هزینه‌ها را تشکیل می‌دهد، در حالی که لغزش در اجراهای واقعی (به‌ویژه برای معاملات با حجم بالا) غالب است.  
    \item \textbf{تنظیم آنتروپی\footnote{Entropy Regularization}:} افزودن پاداش آنتروپی باعث بهبود تنوع‌سازی می‌شود:  
    \begin{itemize}
        \item $\eta = 0.05$ بالاترین بازده سالانه\footnote{Annualized Return, ARR} را به دست می‌آورد.  
        \item $\eta = 0.1$ کمترین افت سرمایه بیشینه\footnote{Maximum Drawdown, MDD} را حاصل می‌کند.  
    \end{itemize}
    \item \textbf{رفتار اجرا:} سیاست سطح پایین معاملات را به‌صورت راهبردی پخش می‌کند، مانند فروش جزئی زودهنگام و هم‌تراز کردن خرید/فروش با پویایی کوتاه‌مدت قیمت، که هزینه‌های لغزش را کاهش می‌دهد.  
    \item \textbf{عملکرد کلی:} HRPM در هر دو بازار ایالات متحده و چین بهتر از خط‌مبناهای DRL و راهبردهای کلاسیک پرتفوی عمل می‌کند.  
\end{itemize}

    \item حساسیت به ابرپارامترها (مانند وزن آنتروپی $\eta$) بر تعادل میان ریسک و بازده تأثیر می‌گذارد.

این مقاله به محدودیت‌های مقاومت مدل‌های DRL تک‌عاملی\footnote{Deep Reinforcement Learning, DRL} در معامله‌گری مالی می‌پردازد. نویسندگان \textbf{MADDQN} را معرفی می‌کنند، یک چارچوب یادگیری تقویتی عمیق چندعاملی\footnote{Multi-Agent DRL} که استخراج‌کننده‌های ویژگی متنوعی — \lr{TimesNet} و \lr{Multi-Scale CNN} — را ادغام می‌کند تا راهبردهای معاملاتی با هدف متعادل‌سازی بیشینه‌سازی بازده و کنترل ریسک توسعه دهد.  

\subsubsection{روش‌شناسی}
چارچوب پیشنهادی یک طراحی دو‌عاملی را به‌کار می‌گیرد که هرکدام در استخراج ویژگی متفاوتی تخصص دارند:
\begin{itemize}
    \item \textbf{عامل ۱:} از \lr{TimesNet} استفاده می‌کند که سری‌های زمانی یک‌بعدی را به نمایش‌های دوبعدی تبدیل می‌کند تا وابستگی‌های زمانی دوره‌ای و چندمقیاسی را ثبت کند.  
    \item \textbf{عامل ۲:} از \lr{Multi-Scale CNN} برای استخراج ویژگی‌های زمانی در چندین وضوح بهره می‌برد.  
\end{itemize}

\begin{itemize}
    \item \textbf{پیش‌آموزش:} روی یک مجموعه‌داده ترکیبی از شاخص‌های اصلی ایالات متحده برای افزایش قابلیت تعمیم انجام می‌شود.  
    \item \textbf{ریزتنظیم:} روی دارایی‌های منفرد برای انطباق راهبردها با سناریوهای خاص بازار صورت می‌گیرد.  
\end{itemize}

    \item به‌طور مداوم از خط‌مبناهای تک‌عاملی در سودآوری و مقاومت عملکردی بهتر داشت.  
    \item توانایی \textbf{تعمیم‌پذیری} قوی نشان داد؛ پیش‌آموزش قابلیت انتقال را پشتیبانی کرد و ریزتنظیم عملکرد ویژه‌دارایی را تثبیت کرد.  
    \item طراحی چندعاملی پروفایل‌های بازده–ریسک پایدارتر نسبت به سیستم‌های DRL تک‌عاملی ایجاد کرد.  
\end{itemize}

\subsubsection{محدودیت‌ها}
\begin{itemize}
    \item از نظر محاسباتی پرهزینه است به دلیل سربار آموزش و هماهنگ‌سازی چندین عامل.  
    \item اتکا شدید به \lr{TimesNet} و استخراج‌کننده‌های مبتنی بر \lr{CNN} ممکن است انطباق‌پذیری را در رژیم‌های بازار کاملاً متفاوت یا پیش‌بینی‌نشده کاهش دهد.  
    \item ارزیابی به بک‌تست‌های تاریخی محدود بود؛ هیچ آزمایش معاملاتی زنده انجام نشد.  
    \item تمرکز عمدتاً بر شاخص‌های سهام ایالات متحده بوده و تعمیم فرامرزی بررسی نشده است.  
\end{itemize}

\section{رویکردهای ترکیبی: ادغام مدل‌های زبانی بزرگ با یادگیری تقویتی در مالی}

\subsection{FINCON: A Synthesized LLM Multi-Agent System with Conceptual Verbal Reinforcement for Enhanced Financial Decision Making \cite{yu2024fincon}}

\subsubsection{تمرکز پژوهش}
این پژوهش به پیچیدگی تصمیم‌گیری‌های متوالی مالی با ادغام مدل‌های زبانی بزرگ\footnote{Large Language Models, LLMs} و یادگیری تقویتی\footnote{Reinforcement Learning, RL} در یک \textbf{چارچوب چندعاملی}\footnote{Multi-Agent Framework} می‌پردازد. \lr{FINCON} یک سلسله‌مراتب مدیر–تحلیلگر، تقویت کلامی مفهومی\footnote{Conceptual Verbal Reinforcement, CVRF} و کنترل ریسک دو‌سطحی معرفی می‌کند تا مقاومت، تفسیرپذیری و آگاهی از ریسک در راهبردهای مالی خودکار ارتقا یابد.  

\subsubsection{روش‌شناسی}
چارچوب \lr{FINCON} به‌صورت زیر طراحی شده است:
\begin{itemize}
    \item \textbf{معماری چندعاملی:}  
    \begin{itemize}
        \item \textbf{عوامل تحلیلگر بازار:} استخراج و خلاصه‌سازی داده‌های کمی بازار (مانند قیمت سهام و بنیادها).  
        \item \textbf{عوامل خبر و احساسات:} پردازش اخبار مالی، گزارش‌های تحلیلی و شبکه‌های اجتماعی برای تولید بینش‌های احساسی.  
        \item \textbf{عوامل راهبردی:} تدوین اقدامات معاملاتی و پرتفوی پیشنهادی با ادغام ورودی‌های تحلیلگر و احساسات.  
        \item \textbf{عوامل مدیریت ریسک:} اِعمال محدودیت‌های ریسک در سطح پرتفوی، پایش مواجهه و تضمین ایمنی.  
        \item \textbf{عامل مدیر:} تجمیع همه ورودی‌ها، قضاوت میان راهبردها و نهایی‌سازی تصمیمات معاملاتی.  
    \end{itemize}
    \item \textbf{کنترل ریسک:}  
    \begin{itemize}
        \item \textbf{درون‌اپیزودی:} پایش \lr{Conditional Value-at-Risk (CVaR)} برای محدود کردن ریسک کوتاه‌مدت.  
        \item \textbf{میان‌اپیزودی:} به‌روزرسانی باورها و تنظیم راهبردی برای بهبود سازگاری بلندمدت.  
    \end{itemize}
    \item \textbf{CVRF:} عامل‌ها به خودانتقادی پرداخته و «باورهای مفهومی» را به زبان طبیعی بیان می‌کنند تا تصمیم‌گیری‌های آینده هدایت شود.  
    \item \textbf{فرمول‌بندی POMDP\footnote{Partially Observable Markov Decision Process}:} معامله و تخصیص پرتفوی به‌عنوان POMDP مدل‌سازی می‌شوند تا عدم قطعیت‌های کلان اقتصادی و بازار ثبت شوند.  
    \item \textbf{بهینه‌سازی پرامپت:} گرادیان نزولی متنی پرامپت‌ها را میان اپیزودها اصلاح می‌کند تا استدلال و هماهنگی عامل‌ها پایدارتر شود.  
\end{itemize}

\subsubsection{یافته‌ها}
\begin{itemize}
    \item عملکردی بهتر از عامل‌های DRL خط‌مبنا و سایر سیستم‌های LLM-محور در تصمیم‌گیری مالی دارد.  
    \item بهبود در \textbf{بازده تجمعی (CR)}، \textbf{نسبت شارپ (SR)} و کاهش \textbf{بیشینه افت سرمایه (MDD)} را نشان می‌دهد.  
    \item سلسله‌مراتب مدیر–تحلیلگر کارایی را با کاهش سربار ارتباط میان‌عاملی افزایش می‌دهد.  
    \item CVRF تفسیرپذیری و تعمیم‌پذیری را در وظایف تک‌سهمی و مدیریت پرتفوی تقویت می‌کند.  
\end{itemize}

\subsubsection{محدودیت‌ها}
\begin{itemize}
    \item به دلیل پیچیدگی هماهنگی چندعاملی و پردازش چندوجهی ورودی‌ها، محاسبات سنگین و پرهزینه است.  
    \item عملکرد به‌شدت وابسته به مهندسی پرامپت و دسترسی به داده‌های مالی باکیفیت است.  
    \item ارزیابی به بک‌تست تاریخی محدود است؛ عملکرد در معاملات زنده بررسی نشده است.  
    \item مقیاس‌پذیری به شبکه‌های بزرگ‌تر عاملی ممکن است چالش‌های هماهنگی را تشدید کند.  
\end{itemize}
